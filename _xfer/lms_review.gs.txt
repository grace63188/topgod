/**
 * 學院管理後台（topgod）資料中心擴充｜lms_review.gs（v1 新增）
 * 作業兩段審核：助教初審（完整送複審／退回補件）→ 校長複審（通過／需修改）
 *  - 主鍵：課程作業＝mem_work_id；輔導作業＝'c:' + 輔導作業 id
 *  - 退回補件／需修改：寫回兩筆工單（截止日改補件期限、審核狀態改 settings.review.returnWorkStatus〔預設 N〕）＋補件卡入佇列（waiting_wb，工單成功才發）
 *  - 複審通過：寫回審核狀態 Y（輔導作業 approved）
 *  - 學姐重新上傳（同步到的 upload_at 晚於退回時間）→ 自動回到初審
 */
function lmsReviewCfg_() { var d = LMS_SCHEMA.SETTINGS_DEFAULTS.review, s = lmsSettings_().review || {}, o = {}; for (var k in d) o[k] = s[k] !== undefined ? s[k] : d[k]; return o; }
/** 官網作業（works＋cworks）與審核紀錄合併 → 審核列 */
function lmsReviewRows_() {
  var db = lmsDb_(), R = {}, P = {}, M = {};
  lmsRows_('reviews').forEach(function (r) { R[r.id] = r; });
  db.products.forEach(function (p) { P[p.id] = p; });
  db.members.forEach(function (m) { M[m.id] = m; });
  var out = [];
  db.works.forEach(function (w) {
    if (!w.mem_work_id) return;
    var id = String(w.mem_work_id), rv = R[id], site = String(w.status || 'W');
    out.push(lmsReviewRow_(id, 'work', w, rv, site, site === 'Y', !!w.upload_at, P[w.pro_id], M[w.mem_id]));
  });
  var CA = {}; (db.counsels || []).forEach(function (c) { CA[c.id] = c.start_at; });
  db.cworks.forEach(function (c) {
    var id = 'c:' + c.id, rv = R[id], site = String(c.status || 'submitted');
    out.push(lmsReviewRow_(id, 'cwork', c, rv, site, site === 'approved', !!c.upload_at, null, M[c.mem_id], CA[c.counsel_id]));
  });
  return out;
}
function lmsReviewRow_(id, kind, w, rv, site, sitePassed, uploaded, prod, mem, counselAt) {
  /* 收件定義（LMS_SCHEMA.reviewQueueOf）：近 N 天（設定 review.windowDays，預設 30）且仍需處理 → 初審；較舊仍未審 → 歷史未審（history）；官網已審／課程已過期／諮詢已結束 → 不列 */
  var q = LMS_SCHEMA.reviewQueueOf({ upload_at: w.upload_at, status: kind === 'cwork' ? (w.status || '') : (w.status || ''), expired_at: w.expired_at, kind: kind, cKind: w.c_kind, counsel_at: counselAt || w.counsel_at, deadline_at: w.deadline_at, seen: !!(rv && rv.seen) }, { windowDays: lmsReviewCfg_().windowDays });
  var stage = rv && rv.stage ? rv.stage : (sitePassed ? 'passed' : q === 'pending' ? 'first' : q === 'history' ? 'history' : ''), resub = false;
  if (rv && (stage === 'returned' || stage === 'revise') && w.upload_at && rv.returnedAt && LMS_CORE.parse(w.upload_at) > LMS_CORE.parse(rv.returnedAt)) { stage = 'first'; resub = true; }
  if (!(rv && rv.stage) && sitePassed) stage = 'passed';
  return { id: id, kind: kind, memId: Number(w.mem_id), memberName: mem ? mem.name : '', level: mem ? mem.level : '', proId: kind === 'work' ? w.pro_id : '', proName: kind === 'work' ? (w.pro_name || (prod && prod.name) || '') : (w.counsel_title || ''),
    workType: kind === 'work' ? (w.work_type || '') : 'C', title: kind === 'work' ? (w.work_title || LMS_CORE.WORK_TYPES[w.work_type] || '作業') : (w.title || '輔導作業'), uploadAt: w.upload_at || '', deadlineAt: w.deadline_at || '', siteStatus: site,
    memProId: w.mem_pro_id || '', workId: w.work_id || '', stage: stage, first: rv ? rv.first || null : null, final: rv ? rv.final || null : null, returnDeadline: rv ? rv.returnDeadline || '' : '', returnedAt: rv ? rv.returnedAt || '' : '',
    wbIds: rv ? rv.wbIds || [] : [], cardIds: rv ? rv.cardIds || [] : [], resubmitted: resub, queue: q, seen: !!(rv && rv.seen), history: rv ? rv.history || [] : [] };
}
/** lms_review_list { stage?, type?, memId? } */
function lmsReviewList_(req) {
  var rows = lmsReviewRows_().filter(function (r) { return r.stage; }), counts = {};
  rows.forEach(function (r) { var k = r.stage; counts[k] = (counts[k] || 0) + 1; var t = 'type:' + r.workType + ':' + k; counts[t] = (counts[t] || 0) + 1; });
  if (req.stage) rows = rows.filter(function (r) { return r.stage === req.stage; });
  if (req.type) rows = rows.filter(function (r) { return r.workType === req.type; });
  if (req.memId) rows = rows.filter(function (r) { return r.memId === Number(req.memId); });
  rows.sort(function (a, b) { return String(a.uploadAt || a.deadlineAt).localeCompare(String(b.uploadAt || b.deadlineAt)); });
  return { rows: rows.slice(0, 800), counts: counts };
}
function lmsReviewFind_(id) {
  id = String(id || ''); var r = lmsReviewRows_().filter(function (x) { return x.id === id; })[0];
  if (!r) throw apiError('找不到這份作業（請先同步）', 'ERR_NOT_FOUND');
  return r;
}
function lmsReviewSave_(row, patch, me, what) {
  var rv = lmsGet_('reviews', row.id) || { id: row.id, kind: row.kind, memId: row.memId, stage: row.stage, wbIds: [], cardIds: [], history: [] };
  for (var k in patch) rv[k] = patch[k];
  rv.updatedAt = nowIso();
  rv.history = (rv.history || []).concat([{ at: nowIso(), by: me.email, role: me.role, what: what, note: patch.first && what.indexOf('first') === 0 ? patch.first.note : patch.final ? patch.final.note : '' }]).slice(-40);
  lmsPut_('reviews', [rv]);
  return rv;
}
/** 退回／需修改：兩筆工單（截止日、審核狀態）＋補件卡（等工單成功才發） */
function lmsReviewReturn_(row, newDeadline, missing, note, me, n) {
  var cfg = lmsReviewCfg_(), specs = [], src = { type: 'review', id: row.id };
  if (cfg.returnSetsDeadline) {
    if (!newDeadline || isNaN(LMS_CORE.parse(newDeadline))) throw apiError('請設定補件期限', 'ERR_VALIDATION');
    if (LMS_CORE.parse(newDeadline) < Date.now()) throw apiError('補件期限不能早於現在', 'ERR_VALIDATION');
  }
  var dl = newDeadline ? LMS_CORE.iso(newDeadline) : '';
  if (row.kind === 'work') {
    if (cfg.returnSetsDeadline) specs.push({ type: 'work_deadline', memId: row.memId, params: { memWorkId: row.id, memProId: row.memProId, work_id: row.workId, newValue: dl, oldValue: row.deadlineAt || null }, source: src, idemKey: 'review:' + row.id + ':' + n + ':deadline', label: '作業截止日｜' + row.proName });
    specs.push({ type: 'work_status', memId: row.memId, params: { memWorkId: row.id, newValue: cfg.returnWorkStatus, oldValue: row.siteStatus || null }, source: src, idemKey: 'review:' + row.id + ':' + n + ':status', label: '作業審核狀態｜' + row.proName });
  } else {
    var cid = row.id.replace(/^c:/, '');
    if (cfg.returnSetsDeadline) specs.push({ type: 'counsel_work', memId: row.memId, params: { cworkId: cid, newValue: dl, oldValue: row.deadlineAt || null }, source: src, idemKey: 'review:' + row.id + ':' + n + ':deadline', label: '輔導作業截止日｜' + row.title });
    specs.push({ type: 'counsel_work_status', memId: row.memId, params: { cworkId: cid, newValue: cfg.returnCworkStatus, oldValue: row.siteStatus || null }, source: src, idemKey: 'review:' + row.id + ':' + n + ':status', label: '輔導作業審核｜' + row.title });
  }
  var jobs = lmsWbCreateMany_(specs, me).map(function (x) { return x.job; });
  var S = lmsSettings_(), link = row.kind === 'work' && row.proId ? LMS_CORE.fillUrl(S.links.work, { pro_id: row.proId }) : S.links.home;
  var miss = (missing || []).map(function (m) { return cleanText(m).slice(0, 40); }).filter(Boolean).slice(0, 8);
  var card = { kind: 'homework', memId: row.memId, title: '{course}需要補件', vars: { course: row.proName || row.title, date: dl ? LMS_CORE.fmtD(dl) : '' },
    body: '{name}學姐好，{course}的' + row.title + '還差一點點就完成了🤍\n' + (miss.length ? '請補上：**' + miss.join('、') + '**' : '請依助教說明補件') + (dl ? '，並在 **' + LMS_CORE.fmtDT(dl) + ' 前**重新上傳。' : '。') + (note ? '\n' + cleanText(note).slice(0, 200) : ''),
    info: { kind: 'slip', title: '補件資訊', right: 'RESUBMIT', rows: [['作業', row.title], ['補件期限', dl ? LMS_CORE.fmtDT(dl) : '—'], ['需補項目', miss.length ? miss.join('、') : '依助教說明']] },
    buttons: [{ slot: 'primary', label: '重新上傳', dest: link }, { slot: 'secondary', label: '我想問問題', dest: S.links.home }], source: { type: 'review', id: row.id }, waitWb: jobs.map(function (j) { return j.id; }) };
  var sent = lmsModuleSend_([card], me, {});
  return { jobs: jobs, cardIds: sent.queued, card: sent };
}
/** lms_review_first { memWorkId, result:'pass'|'return', missing, note, newDeadline? }（助教或校長） */
function lmsReviewFirst_(req, me) {
  var row = lmsReviewFind_(req.memWorkId), result = String(req.result || '');
  if (LMS_SCHEMA.ENUMS.REVIEW_FIRST_RESULT.indexOf(result) < 0) throw apiError('初審結果只能是 pass 或 return', 'ERR_VALIDATION');
  if (row.stage !== 'first') throw apiError('這份作業目前在「' + (LMS_SCHEMA.ENUMS.REVIEW_STAGE_LABEL[row.stage] || row.stage || '未上傳') + '」，不能初審', 'ERR_STATE');
  var first = { by: me.email, name: me.name, at: nowIso(), result: result, missing: (req.missing || []).slice(0, 12).map(function (m) { return cleanText(m).slice(0, 60); }), note: cleanText(req.note).slice(0, 1000) };
  if (result === 'pass') return { review: lmsReviewSave_(row, { stage: 'final', first: first, resubmittedAt: row.resubmitted ? row.uploadAt : '' }, me, 'first_pass'), jobs: [], card: null };
  var n = ((lmsGet_('reviews', row.id) || {}).history || []).filter(function (h) { return /return|revise/.test(h.what); }).length + 1;
  var r = lmsReviewReturn_(row, req.newDeadline, first.missing, first.note, me, n);
  var rv0 = lmsGet_('reviews', row.id) || {};
  var rv = lmsReviewSave_(row, { stage: 'returned', first: first, returnDeadline: req.newDeadline ? LMS_CORE.iso(req.newDeadline) : '', returnedAt: nowIso(), wbIds: (rv0.wbIds || []).concat(r.jobs.map(function (j) { return j.id; })), cardIds: (rv0.cardIds || []).concat(r.cardIds) }, me, 'first_return');
  return { review: rv, jobs: r.jobs, card: r.card };
}
/** lms_review_final { memWorkId, result:'pass'|'revise', note, newDeadline? }（校長） */
function lmsReviewFinal_(req, me) {
  var row = lmsReviewFind_(req.memWorkId), result = String(req.result || ''), cfg = lmsReviewCfg_();
  if (LMS_SCHEMA.ENUMS.REVIEW_FINAL_RESULT.indexOf(result) < 0) throw apiError('複審結果只能是 pass 或 revise', 'ERR_VALIDATION');
  if (row.stage !== 'final') throw apiError('這份作業還沒送複審', 'ERR_STATE');
  var fin = { by: me.email, name: me.name, at: nowIso(), result: result, note: cleanText(req.note).slice(0, 1000) }, rv0 = lmsGet_('reviews', row.id) || {};
  if (result === 'pass') {
    var spec = row.kind === 'work'
      ? { type: 'work_status', memId: row.memId, params: { memWorkId: row.id, newValue: cfg.passWorkStatus, oldValue: row.siteStatus || null }, source: { type: 'review', id: row.id }, idemKey: 'review:' + row.id + ':pass', label: '作業審核狀態｜' + row.proName }
      : { type: 'counsel_work_status', memId: row.memId, params: { cworkId: row.id.replace(/^c:/, ''), newValue: cfg.passCworkStatus, oldValue: row.siteStatus || null }, source: { type: 'review', id: row.id }, idemKey: 'review:' + row.id + ':pass', label: '輔導作業審核｜' + row.title };
    var jobs = lmsWbCreateMany_([spec], me).map(function (x) { return x.job; });
    return { review: lmsReviewSave_(row, { stage: 'passed', final: fin, wbIds: (rv0.wbIds || []).concat(jobs.map(function (j) { return j.id; })) }, me, 'final_pass'), jobs: jobs, card: null };
  }
  var n = (rv0.history || []).filter(function (h) { return /return|revise/.test(h.what); }).length + 1;
  var r = lmsReviewReturn_(row, req.newDeadline, [], fin.note, me, n);
  var rv = lmsReviewSave_(row, { stage: 'revise', final: fin, returnDeadline: req.newDeadline ? LMS_CORE.iso(req.newDeadline) : '', returnedAt: nowIso(), wbIds: (rv0.wbIds || []).concat(r.jobs.map(function (j) { return j.id; })), cardIds: (rv0.cardIds || []).concat(r.cardIds) }, me, 'final_revise');
  return { review: rv, jobs: r.jobs, card: r.card };
}
var LMS_ACTIONS_REVIEW = {
  lms_review_list: function (req) { return lmsReviewList_(req); },
  lms_review_first: function (req, me) { return lmsReviewFirst_(req, me); },
  lms_review_final: function (req, me) { return lmsReviewFinal_(req, me); }
};
