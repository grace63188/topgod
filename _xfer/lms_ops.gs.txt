/**
 * 學院管理後台（topgod）資料中心擴充｜lms_ops.gs（契約 v1.1 新增，只新增動作與欄位）
 *  - lms_workbench：工作台彙總（四格 KPI、五站待辦、截止三卡、今日諮詢準備度、最近動態）
 *  - lms_quota_status：line_quota_status（consult-prep，不改）＋天使 OA＋本月每週用量
 *  - 同步紀錄：每次同步一列 SY-yyyymmdd-nn（讀取、新增、變更、事件、錯誤、逐筆差異）；lms_sync_state.runs／today／assistantBmVersion、lms_sync_runs
 *  - 逐人同步範圍：settings.syncScope（list＝指定名單）或啟用且級別未到期；沒有名單時不帶 perMemberIds（書籤自行判斷）
 *  - 待審核（工單以外）：派發批次、課表定案、補交核准；寫回今日統計
 *  - lms_settings_get.users[].lastLogin；lms_dump members 推導欄（expiring_points、renewed_at、topup_at）
 *  - lms_state_put 已知鍵檢查（consult:、consult_cancel:、consult_stop:、plan_apply:）
 */
var LMS_OPS = { RUN_GAP_MS: 20 * 60000, RUNS_KEEP: 30, DIFF_MAX: 40, RECENT_DAYS: 30 };

/* ── 逐人同步範圍 ─────────────────────────────────────── */
/** → { ids:[memId]|null, source:'list'|'auto'|'' }；ids 為 null＝不帶名單（書籤依同一規則自行判斷） */
function lmsPerMemberScope_() {
  try {
    var sc = lmsSettings_().syncScope || {};
    if (sc.mode === 'list' && Array.isArray(sc.memberIds) && sc.memberIds.length) return { ids: sc.memberIds.map(Number).filter(function (x) { return x > 0; }), source: 'list' };
    var now = Date.now();
    var ids = lmsRows_('members').filter(function (m) { return (m.status || 'Y') === 'Y' && (!m.expired_at || LMS_CORE.parse(m.expired_at) >= now); }).map(function (m) { return Number(m.id); }).filter(function (x) { return x > 0; });
    return ids.length ? { ids: ids, source: 'auto' } : { ids: null, source: '' };
  } catch (e) { return { ids: null, source: '' }; }
}

/* ── 同步紀錄（SY-yyyymmdd-nn） ───────────────────────── */
function lmsRunScopeLabel_(k) { return (LMS_SCHEMA.ENUMS.SYNC_SCOPE || {})[k] || ''; }
function lmsRunNewId_() {
  var day = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyyMMdd'), pre = 'SY-' + day + '-', max = 0;
  lmsRows_('sync_runs').forEach(function (r) { if (String(r.id).indexOf(pre) === 0) max = Math.max(max, Number(String(r.id).slice(pre.length)) || 0); });
  var n = max + 1; return pre + (n < 10 ? '0' + n : String(n));
}
/** 開新同步紀錄（書籤 op:'start'；或距上一批超過 20 分鐘、或後台送來不同的 runId） */
function lmsRunOpen_(me, opt) {
  opt = opt || {};
  var st = lmsGet_('sync_runs', 'state') || { id: 'state', counts: {} };
  var r = { id: lmsRunNewId_(), at: nowIso(), lastAt: nowIso(), by: me ? me.email : '', role: (opt.role === 'assistant' || opt.role === 'principal') ? opt.role : (me && me.realRole === 'assistant' ? 'assistant' : (me && me.via === 'syncKey' ? '' : 'principal')),
    scopeKey: LMS_SCHEMA.ENUMS.SYNC_SCOPE[opt.scope] ? opt.scope : '', ext: String(opt.runId || '').slice(0, 60), bmVersion: String(opt.bmVersion || '').slice(0, 12),
    kinds: {}, read: 0, added: 0, changed: 0, events: 0, errors: 0, error: '', diff: { works: [], plans: [], counsels: 0, orders: 0, todos: 0 }, closed: '' };
  lmsPut_('sync_runs', [r]);
  st.curRun = { id: r.id, at: r.at, lastAt: r.lastAt, ext: r.ext };
  lmsPut_('sync_runs', [st]);
  return r;
}
/** 目前的同步紀錄（必要時開新的） */
function lmsRunCur_(me, req) {
  var st = lmsGet_('sync_runs', 'state') || {}, c = st.curRun, ext = String((req && req.runId) || '');
  var r = c && c.id ? lmsGet_('sync_runs', c.id) : null;
  var stale = !r || r.closed || (Date.now() - LMS_CORE.parse(r.lastAt || r.at) > LMS_OPS.RUN_GAP_MS) || (ext && r.ext && ext !== r.ext);
  if (stale) r = lmsRunOpen_(me, { runId: ext, scope: req && req.meta && req.meta.scope, bmVersion: req && req.meta && req.meta.bmVersion });
  else if (ext && !r.ext) r.ext = ext;
  return r;
}
function lmsRunWho_(memId) { var m = lmsGet_('members', memId); return m ? m.name : ''; }
function lmsRunTs_(v) { return v ? LMS_CORE.fmtDT(v) : '空白'; }
/**
 * 同步一批後記入同步紀錄：old＝寫入前的同 id 列（{id: row}）；clean＝白名單後的新列
 *  works：上傳時間、審核狀態變更；site_plans：新申請、狀態變更；counsels／orders：新增筆數
 */
function lmsRunTrack_(kind, old, clean, received, req, me) {
  try {
    var r = lmsRunCur_(me, req), added = 0, changed = 0, D = r.diff = r.diff || { works: [], plans: [], counsels: 0, orders: 0, todos: 0 }, ST = LMS_SCHEMA.ENUMS.SITE_WORK_STATUS;
    clean.forEach(function (o) {
      var b = old[String(o.id)];
      if (!b) added++; else if (JSON.stringify(b) !== JSON.stringify(o)) changed++; else return;
      if (kind === 'works' && o.mem_work_id) {
        if (b && String(b.upload_at || '') !== String(o.upload_at || '')) D.works.push({ who: lmsRunWho_(o.mem_id), memId: Number(o.mem_id), what: (o.pro_name || '') + ' ' + (o.work_title || ''), from: '上傳時間 ' + lmsRunTs_(b.upload_at), to: lmsRunTs_(o.upload_at) });
        if (b && String(b.status || '') !== String(o.status || '')) D.works.push({ who: lmsRunWho_(o.mem_id), memId: Number(o.mem_id), what: (o.pro_name || '') + ' ' + (o.work_title || ''), from: ST[b.status] || b.status || '未審核', to: ST[o.status] || o.status || '未審核' });
      }
      if (kind === 'site_plans') {
        if (!b) D.plans.push({ who: lmsRunWho_(o.mem_id), memId: Number(o.mem_id), what: '個人進修課表', from: '無', to: (o.status === 'pending' ? '官網申請 ' : o.status + ' ') + LMS_CORE.fmtD(o.created_at) });
        else if (String(b.status || '') !== String(o.status || '')) D.plans.push({ who: lmsRunWho_(o.mem_id), memId: Number(o.mem_id), what: '個人進修課表', from: b.status || '—', to: o.status || '—' });
      }
      if (kind === 'counsels' && !b) D.counsels++;
      if (kind === 'orders' && !b) D.orders++;
    });
    D.works = D.works.slice(-LMS_OPS.DIFF_MAX); D.plans = D.plans.slice(-LMS_OPS.DIFF_MAX);
    r.kinds = r.kinds || {}; r.kinds[kind] = (Number(r.kinds[kind]) || 0) + received;
    r.read = (Number(r.read) || 0) + received; r.added = (Number(r.added) || 0) + added; r.changed = (Number(r.changed) || 0) + changed;
    r.events = D.works.length + D.plans.length + D.counsels + D.orders;
    if (req && req.meta && req.meta.error) { r.errors = (Number(r.errors) || 0) + 1; r.error = cleanText(req.meta.error).slice(0, 200); }
    if (req && req.meta && req.meta.scope && LMS_SCHEMA.ENUMS.SYNC_SCOPE[req.meta.scope]) r.scopeKey = req.meta.scope;
    r.lastAt = nowIso(); if (!r.by && me) r.by = me.email;
    if (me && me.via !== 'syncKey' && !r.role) r.role = me.realRole === 'assistant' ? 'assistant' : 'principal';
    lmsPut_('sync_runs', [r]);
    var st = lmsGet_('sync_runs', 'state'); if (st && st.curRun && st.curRun.id === r.id) { st.curRun.lastAt = r.lastAt; lmsPut_('sync_runs', [st]); }
    return r;
  } catch (e) { console.error('lms run track ' + e.message); return null; }
}

/** 書籤 op:'done'：結束同步紀錄 */
function lmsRunClose_(errors) {
  var st = lmsGet_('sync_runs', 'state') || {}, r = st.curRun && st.curRun.id ? lmsGet_('sync_runs', st.curRun.id) : null;
  if (!r || r.closed) return null;
  var errs = (Array.isArray(errors) ? errors : []).map(function (e) { return cleanText(e).slice(0, 200); }).filter(Boolean).slice(0, 10);
  if (errs.length) { r.errors = (Number(r.errors) || 0) + errs.length; r.error = errs.join('；').slice(0, 300); }
  r.closed = nowIso(); r.lastAt = r.closed;
  lmsPut_('sync_runs', [r]);
  return lmsRunPublic_(r);
}
function lmsRunPublic_(r) {
  var D = r.diff || {}, todos = 0;
  try { var a = LMS_CORE.parse(r.at), b = LMS_CORE.parse(r.closed || r.lastAt || r.at) + 5 * 60000; lmsRows_('todos').forEach(function (t) { var x = LMS_CORE.parse(t.at); if (x >= a && x <= b) todos++; }); } catch (e) { todos = 0; }
  var scope = lmsRunScopeLabel_(r.scopeKey) || ((r.kinds || {}).products || (r.kinds || {}).kinds ? '完整' : '快速');
  var byName = r.role === 'assistant' ? '助教' : r.role === 'principal' ? '校長' : (r.by === 'bookmarklet' ? '書籤' : (r.by || ''));
  var out = { id: r.id, at: r.at, by: byName, email: r.by || '', role: r.role || '', scope: scope, read: Number(r.read) || 0, added: Number(r.added) || 0, changed: Number(r.changed) || 0, events: (Number(r.events) || 0), errors: Number(r.errors) || 0, kinds: r.kinds || {}, closed: r.closed || '', bmVersion: r.bmVersion || '',
    diff: { works: D.works || [], plans: D.plans || [], counsels: D.counsels || 0, orders: D.orders || 0, todos: todos } };
  if (r.error) out.error = r.error;
  return out;
}
function lmsRunsList_(limit) {
  var rows = lmsRows_('sync_runs').filter(function (r) { return /^SY-\d{8}-\d+$/.test(String(r.id)); });
  var ord = function (r) { var m = /^SY-(\d{8})-(\d+)$/.exec(String(r.id)); return m[1] + ('00000' + m[2]).slice(-5); };
  rows.sort(function (a, b) { return ord(b).localeCompare(ord(a)); });   /* 編號依建立順序（同一天遞增） */
  return rows.slice(0, Math.max(1, Math.min(100, Number(limit) || LMS_OPS.RUNS_KEEP))).map(lmsRunPublic_);
}
function lmsRunsToday_(runs) {
  var today = LMS_CORE.dayKey(Date.now()), wk = Date.now() - 7 * 86400000, o = { runs: 0, quick: 0, full: 0, member: 0, read: 0, errors: 0, errors7d: 0 };
  runs.forEach(function (r) {
    if (LMS_CORE.parse(r.at) >= wk) o.errors7d += r.errors;
    if (LMS_CORE.dayKey(r.at) !== today) return;
    o.runs++; o.read += r.read; o.errors += r.errors;
    if (r.scope === '完整') o.full++; else if (r.scope === '指定學姐') o.member++; else o.quick++;
  });
  return o;
}

function lmsWbToday_(jobs) {
  var today = LMS_CORE.dayKey(Date.now()), o = { done: 0, failed: 0, manual: 0, conflict: 0, lastAt: '' };
  jobs.forEach(function (j) {
    (j.history || []).forEach(function (h) {
      if (LMS_CORE.dayKey(h.at) !== today) return;
      if (h.what === 'done' || h.what === 'failed' || h.what === 'manual' || h.what === 'conflict') { o[h.what]++; if (h.what !== 'manual' && String(h.at) > o.lastAt) o.lastAt = h.at; }
    });
  });
  return o;
}

/* ── lms_dump：學姐推導欄（不存） ─────────────────────── */
function lmsMembersDerived_(members) {
  var now = Date.now(), since = now - LMS_OPS.RECENT_DAYS * 86400000, ren = {}, top = {};

/* ── lms_state_put 已知鍵檢查 ─────────────────────────── */
function lmsStateRowCheck_(id, row) {
  var r = row && typeof row === 'object' ? LMS_SCHEMA.stripSensitive(row) : row;
  if (id.indexOf('consult_cancel:') === 0) {
    if (!/^consult_cancel:\d+$/.test(id)) throw apiError('consult_cancel 鍵需為 consult_cancel:<諮詢 id>', 'ERR_VALIDATION');
    if (r && r.stage != null && LMS_SCHEMA.ENUMS.CONSULT_CANCEL_STAGE.indexOf(r.stage) < 0) throw apiError('取消補交階段只能是 ' + LMS_SCHEMA.ENUMS.CONSULT_CANCEL_STAGE.join('、'), 'ERR_VALIDATION');
  } else if (id.indexOf('consult_stop:') === 0) {
    if (!/^consult_stop:\d+$/.test(id)) throw apiError('consult_stop 鍵需為 consult_stop:<學姐 id>', 'ERR_VALIDATION');
    if (r && typeof r === 'object') { r.on = r.on === true; if (r.year != null) r.year = Number(r.year) || ''; }
  } else if (id.indexOf('consult:') === 0) {
    if (!/^consult:\d+$/.test(id)) throw apiError('consult 鍵需為 consult:<諮詢 id>', 'ERR_VALIDATION');
    if (r && typeof r === 'object' && r.summary != null) r.summary = cleanText(r.summary).slice(0, 4000);
  }
  return r;
}

/* ── lms_quota_status ─────────────────────────────────── */
function lmsQuotaStatus_(req) {
  var q = {};
  if (typeof lineQuotaStatus === 'function') { try { q = lineQuotaStatus({ force: req.force === true }) || {}; } catch (e) { q = { error: e.message }; } }
  else { var lq = lmsQuotaSafe_() || {}; for (var k in lq) q[k] = lq[k]; }
  var out = {}; for (var k2 in q) out[k2] = q[k2];
  var S = lmsSettings_(), plan = lmsQuotaPlan_(S);
  out.level = LMS_SCHEMA.quotaLevel(q.remaining, plan); out.siteLevel = q.level;
  out.angel = lmsAngelQuota_();
  var days = {}, dom = Number(Utilities.formatDate(new Date(), 'Asia/Taipei', 'd')), dim = Number(q.daysInMonth) || 30;
  try { if (typeof quotaMonth === 'function') days = quotaMonth().days || {}; } catch (e) { days = {}; }
  var weekly = [], used = 0; Object.keys(days).forEach(function (d) { used += Number(days[d]) || 0; });
  var avg = dom ? used / dom : 0;
  for (var w = 0; w * 7 < dim; w++) {
    var a = w * 7 + 1, b = Math.min(dim, a + 6), n = 0, f = false;
    for (var d = a; d <= b; d++) { var key = ('0' + d).slice(-2); if (d <= dom) n += Number(days[key]) || 0; else { n += avg; f = true; } }
    var row = { w: 'W' + (w + 1), n: Math.round(n), from: a, to: b }; if (f) row.f = true; weekly.push(row);
  }
  out.weekly = weekly; out.forecastWeek = Math.round(avg * 7); out.plan = plan;
  return out;
}

/* ── lms_workbench ───────────────────────────────────── */
function lmsWorkbench_(req, me) {
  var range = ['today', 'week', 'month'].indexOf(req.range) >= 0 ? req.range : 'today';
  /* 2.2.0：彙總結果快取 90 秒（lms_reuse.gs 分段快取；寫入或同步後世代 +1 即失效）；force:true 重新計算 */
  var ck = 'lms_wb_' + range + '_' + ((me && me.role) || 'p');
  if (req.force !== true && typeof lmsCacheGet_ === 'function') { var hit = lmsCacheGet_(ck); if (hit && hit.kpi) { hit.cached = true; return hit; } }
  var out = lmsWorkbenchBuild_(range, me);
  if (typeof lmsCachePut_ === 'function') lmsCachePut_(ck, out, 90);
  return out;
}
function lmsWorkbenchBuild_(range, me) {
  var R = { today: 1, week: 7, month: 30 }[range];
  var now = Date.now(), sod = LMS_CORE.startOfDay(now), from = sod - (R - 1) * 86400000, until = sod + R * 86400000, DAY = 86400000;
  var inR = function (v) { var t = LMS_CORE.parse(v); return !isNaN(t) && t >= from && t < now + 60000; };
  var sis = function (n) { return (n || '') + '學姐'; };
  var reviews = lmsReviewList_({}).rows, applies = lmsPlanApplyList_().rows.filter(function (a) { return !a.closed; });
  var plans = lmsRows_('plans'), ships = lmsRows_('ships'), wb = lmsWbRows_(), queue = lmsRows_('queue'), db = lmsDb_(), stRows = lmsRows_('state');
  var ST = {}; stRows.forEach(function (s) { ST[s.id] = s.row || {}; });
  var first = reviews.filter(function (r) { return r.stage === 'first'; }), fin = reviews.filter(function (r) { return r.stage === 'final'; });
  var resub = first.filter(function (r) { return r.resubmitted; }), firstNew = first.filter(function (r) { return !r.resubmitted; });
  var shipIssue = ships.filter(function (s) { return s.issue_at && !s.issue_resolved_at; });
  var counselsNew = db.counsels.filter(function (c) { return c.status !== 'cancelled' && inR(c.created_at); });
  var sentShip = {}; lmsRows_('log').forEach(function (l) { if (l.kind === 'ship') (l.items || []).forEach(function (i) { sentShip[String(i).replace(/^ship:/, '')] = 1; }); });
  var shipTodo = ships.filter(function (s) { return !s.confirmed_at && !s.notified_at && !sentShip[s.id]; });
  var wrapUp = db.counsels.filter(function (c) { var t = LMS_CORE.parse(c.start_at); return c.status !== 'cancelled' && t < now && t >= now - 14 * DAY && !(ST['consult:' + c.id] || {}).closedAt; });
  var wbPrincipal = wb.filter(function (j) { return j.status === 'pending' && LMS_SCHEMA.wbApproveCap(j.type) === 'wb_approve_money'; });
  var wbBad = wb.filter(function (j) { return j.status === 'failed' || j.status === 'conflict'; });
  var batches = lmsRows_('grant_batches').filter(function (b) { return b.status === 'pending'; });
  var act = lmsActivity_({}).rows, sleep = act.filter(function (a) { return a.tier === '沉睡' || a.tier === '待喚醒'; });
  var waitDays = function (v) { var t = LMS_CORE.parse(v); return isNaN(t) ? 0 : Math.floor((now - t) / DAY); };
  var oldest = function (rows, f) { return rows.slice().sort(function (a, b) { return String(f(a) || '9').localeCompare(String(f(b) || '9')); })[0]; };
  var chip = function (label, n, href) { return [label, n, href]; };
  var st = {};
  var u1 = oldest(firstNew.length ? firstNew : first, function (r) { return r.uploadAt; });
  st.intake = { owner: 'assistant', chips: [chip('作業繳交', firstNew.length, '#/review/first'), chip('課表申請', applies.length, '#/plan/apply'), chip('諮詢預約', counselsNew.length, '#/consult/calendar'), chip('補件回傳', resub.length, '#/review/returned'), chip('包裹異常', shipIssue.length, '#/ship/issue')],
    urgent: u1 ? { title: sis(u1.memberName) + '｜' + (u1.proName ? u1.proName + ' ' : '') + u1.title, l: '已等候 ' + waitDays(u1.uploadAt) + ' 天', r: LMS_CORE.fmtD(u1.uploadAt) + ' 上傳', overdue: waitDays(u1.uploadAt) >= 2, href: '#/review/first/' + encodeURIComponent(u1.id) } : null, cta: { label: '開始初審', href: '#/review/first' } };
  var drafts = plans.filter(function (p) { return p.status === 'draft'; }), d1 = oldest(drafts, function (p) { return p.updated_at; });
  st.make = { owner: 'principal', chips: [chip('排課', drafts.length + applies.filter(function (a) { return a.ok; }).length, '#/plan/schedule'), chip('諮詢收尾', wrapUp.length, '#/consult/records?f=wrap')],
    urgent: d1 ? { title: sis(lmsRunWho_(d1.mem_id)) + '｜個人進修課表排課', l: '更新 ' + LMS_CORE.fmtD(d1.updated_at), r: waitDays(d1.updated_at) ? waitDays(d1.updated_at) + ' 天前' : '今天', overdue: waitDays(d1.updated_at) >= 3, href: '#/plan/schedule?mem=' + d1.mem_id } : null, cta: { label: '處理', href: '#/plan/schedule' } };
  var subm = plans.filter(function (p) { return p.status === 'submitted'; }), f1 = oldest(fin, function (r) { return r.first && r.first.at; });
  st.decide = { owner: 'principal', chips: [chip('作業複審', fin.length, '#/review/final'), chip('課表定案', subm.length, '#/plan/decide'), chip('獎勵派發', batches.length, '#/reward/approve'), chip('工單核准', wbPrincipal.length, '#/workbench/approvals')],
    urgent: f1 ? { title: sis(f1.memberName) + '｜' + (f1.proName ? f1.proName + ' ' : '') + f1.title, l: '初審 ' + LMS_CORE.fmtD(f1.first && f1.first.at), r: '待複審', overdue: waitDays(f1.first && f1.first.at) >= 2, href: '#/review/final/' + encodeURIComponent(f1.id) } : null, cta: { label: '處理', href: '#/workbench/approvals' } };
  var decided = plans.filter(function (p) { return p.status === 'decided'; }), qOpen = queue.filter(function (q) { return q.status === 'scheduled' || q.status === 'waiting_wb' || q.status === 'paused'; });
  var q1 = qOpen.slice().sort(function (a, b) { return String(a.sendAt).localeCompare(String(b.sendAt)); })[0];
  st.send = { owner: 'assistant', chips: [chip('課表發布', decided.length, '#/plan/published'), chip('寄送', shipTodo.length, '#/ship/pending'), chip('提醒', qOpen.length, '#/templates/queue')],
    urgent: q1 ? { title: ((q1.card && q1.card.memberName) ? sis(q1.card.memberName) + '｜' : '') + ((q1.card && q1.card.title) || '通知'), l: '排程 ' + LMS_CORE.fmtDT(q1.sendAt), r: q1.status === 'waiting_wb' ? '等寫回' : q1.status === 'paused' ? '已暫停' : (LMS_CORE.dayKey(q1.sendAt) === LMS_CORE.dayKey(now) ? '今天' : LMS_CORE.fmtD(q1.sendAt)), overdue: q1.status === 'scheduled' && LMS_CORE.parse(q1.sendAt) < now - 3600000, href: '#/templates/queue?sel=' + encodeURIComponent(q1.id) } : null, cta: { label: '處理', href: '#/templates/queue' } };
  var s1 = sleep.slice().sort(function (a, b) { return (b.idleDays || 0) - (a.idleDays || 0); })[0];
  var histN = reviews.filter(function (r) { return r.stage === 'history'; }).length;   /* 較舊仍未審：只在關注提示，不算今日待辦 */
  st.watch = { owner: 'principal', chips: [chip('IG 互動', sleep.length, '#/student/ig'), chip('異常', wbBad.length, '#/workbench/writeback?status=failed')], extra: [['歷史未審', histN, '#/review/history']],
    urgent: wbBad.length ? { title: (wbBad[0].memberName ? sis(wbBad[0].memberName) + '｜' : '') + wbBad[0].label, l: LMS_SCHEMA.ENUMS.WB_STATUS_LABEL[wbBad[0].status], r: (wbBad[0].result && wbBad[0].result.error) || '', overdue: true, href: '#/workbench/writeback/' + encodeURIComponent(wbBad[0].id) }
      : s1 ? { title: sis(s1.name) + '｜' + s1.tier + (s1.idleDays != null ? ' ' + s1.idleDays + ' 天' : ''), l: '建議 IG 互動', r: '本週', overdue: false, href: '#/student/activity?sel=' + s1.memId } : null, cta: { label: '處理', href: '#/student/ig' } };
  Object.keys(st).forEach(function (k) { st[k].count = st[k].chips.reduce(function (a, c) { return a + (Number(c[1]) || 0); }, 0); });
  var total = 0; Object.keys(st).forEach(function (k) { total += st[k].count; });
  var doneN = lmsRows_('audit').filter(function (a) { return inR(a.at) && !/^login_/.test(a.action); }).length;
  var overdue = firstNew.filter(function (r) { return waitDays(r.uploadAt) >= 2; }).length + wbBad.length + reviews.filter(function (r) { return (r.stage === 'returned' || r.stage === 'revise') && r.returnDeadline && LMS_CORE.parse(r.returnDeadline) < now; }).length;
  /* 諮詢準備度 */
  var cw = {}; db.cworks.forEach(function (c) { (cw[c.counsel_id] = cw[c.counsel_id] || []).push(c); });
  var zoomOk = {}; wb.forEach(function (j) { if (j.type === 'zoom_create' && (j.status === 'done' || j.status === 'dryrun')) zoomOk[j.source && j.source.id] = 1; });
  var sess = db.counsels.filter(function (c) { var t = LMS_CORE.parse(c.start_at); return c.status !== 'cancelled' && t >= sod && t < until; }).sort(function (a, b) { return String(a.start_at).localeCompare(String(b.start_at)); });
  var nextSet = false;
  var consults = sess.slice(0, 20).map(function (c) {
    var ws = cw[c.id] || [], s0 = ST['consult:' + c.id] || {};
    var prep = [[ws.length ? (ws.every(function (w) { return w.upload_at; }) ? '作業' : '輔導單未交') : '作業', !ws.length || ws.every(function (w) { return w.upload_at; })], [(s0.zoomCardAt || zoomOk[String(c.id)]) ? 'Zoom' : 'Zoom 未建', !!(s0.zoomCardAt || zoomOk[String(c.id)])]];
    var ok = prep.filter(function (p) { return p[1]; }).length, name = lmsRunWho_(c.mem_id);
    var o = { id: c.id, memId: Number(c.mem_id), at: c.start_at, time: LMS_CORE.fmtT(c.start_at), name: name, av: String(name || '').slice(-1), topic: c.type_name || '', prep: prep, ready: Math.round(ok / prep.length * 100) };
    if (!nextSet && LMS_CORE.parse(c.start_at) >= now) { o.next = true; nextSet = true; }
    if (!prep[0][1]) o.action = { label: '催輔導單', href: '#/consult/prep?sel=' + c.id }; else if (!prep[1][1]) o.action = { label: '建 Zoom', href: '#/consult/detail/' + c.id };
    return o;
  });
  var nx = consults.filter(function (c) { return c.next; })[0];
  /* 截止三卡 */
  var dd = lmsDeadlines_({ days: range === 'month' ? 30 : 7 }), its = dd.items;
  var hasNode = function (x, s) { return (x.nodes || []).some(function (n) { return n.status === s; }); };
  var row3 = function (x, href) { return [sis(lmsRunWho_(x.mem_id)) + '｜' + (x.course || x.title), x.at, href + (href.indexOf('?') >= 0 ? '&' : '?') + 'mem=' + x.mem_id]; };
  var view = its.filter(function (x) { return x.kind === 'watch'; }), acct = its.filter(function (x) { return x.kind === 'level' || x.kind === 'points'; }), hw = its.filter(function (x) { return ['work', 'gwork', 'cwork'].indexOf(x.kind) >= 0; });
  var allHw = lmsItems_(true).filter(function (x) { return ['work', 'gwork', 'cwork'].indexOf(x.kind) >= 0 && LMS_CORE.inRange(x, range === 'month' ? 30 : 7, now); });
  var dl = { viewing: { n: view.length, reminded: view.filter(function (x) { return hasNode(x, 'sent'); }).length, rows: view.slice(0, 3).map(function (x) { return row3(x, '#/deadline/viewing'); }) },
    account: { level: acct.filter(function (x) { return x.kind === 'level'; }).length, points: acct.filter(function (x) { return x.kind === 'points'; }).length, rows: acct.slice(0, 3).map(function (x) { return row3(x, '#/deadline/account'); }) },
    homework: { n: hw.length + allHw.filter(function (x) { return x.done; }).length, submitted: allHw.filter(function (x) { return x.done; }).length, rows: hw.slice(0, 3).map(function (x) { return row3(x, '#/deadline/homework'); }) } };
  var nonShip = its.filter(function (x) { return x.kind !== 'ship'; }), sched = nonShip.filter(function (x) { return hasNode(x, 'pending') || hasNode(x, 'sent'); }).length;
  /* 最近動態 */
  var feed = [];
  reviews.filter(function (r) { return r.uploadAt && LMS_CORE.parse(r.uploadAt) >= now - 3 * DAY; }).forEach(function (r) { feed.push({ ic: 'mail', c: 'work', text: sis(r.memberName) + '繳交 ' + (r.proName ? r.proName + ' ' : '') + r.title, at: r.uploadAt, href: '#/review/first/' + encodeURIComponent(r.id) }); });
  lmsRows_('log').filter(function (l) { return !l.skipped && l.sentAt && LMS_CORE.parse(l.sentAt) >= now - 3 * DAY; }).forEach(function (l) { feed.push({ ic: 'cards', c: 'notice', text: (l.memberName ? sis(l.memberName) + '｜' : '') + (l.title || '通知') + (l.clickAt ? '（已點擊）' : ''), at: l.sentAt, href: '#/templates/detail/' + encodeURIComponent(l.id) }); });
  applies.forEach(function (a) { if (LMS_CORE.parse(a.createdAt) >= now - 3 * DAY) feed.push({ ic: 'wheel', c: 'plan', text: sis(a.memberName) + '送出 進修課表申請', at: a.createdAt, href: '#/plan/apply?sel=' + a.sitePlanId }); });
  lmsRunsList_(5).forEach(function (r) { if (LMS_CORE.parse(r.at) >= now - 3 * DAY) feed.push({ ic: 'sync', c: 'sync', text: '同步官網資料（' + r.scope + '）讀取 ' + r.read + ' 筆、變更 ' + r.changed + ' 筆' + (r.errors ? '、錯誤 ' + r.errors : ''), at: r.at, href: '#/workbench/sync/' + r.id }); });
  feed.sort(function (a, b) { return LMS_CORE.parse(b.at) - LMS_CORE.parse(a.at); });
  var activeN = act.filter(function (a) { return a.tier === '亮眼' || a.tier === '穩定'; }).length;
  return { range: range, from: LMS_CORE.iso(from), generatedAt: nowIso(),
    /* 今日待辦：open＝五站待辦合計（同一份 st 計數，副標、儀表、五站一致）；overdue＝超過處理時限；done＝今日已處理（操作紀錄，獨立正數，不相減） */
    kpi: { today: { total: total, open: total, done: doneN, overdue: overdue, delta: 0 }, sessions: { total: sess.length, ready: consults.filter(function (c) { return c.ready === 100; }).length, pending: consults.filter(function (c) { return c.ready < 100; }).length, next: nx ? nx.time : '' },
      deadlines: { total: nonShip.length, scheduled: sched, unscheduled: nonShip.length - sched, viewing: view.length, account: acct.length, homework: hw.length, days: range === 'month' ? 30 : 7 },
      active: { count: activeN, rate: act.length ? Math.round(activeN / act.length * 100) : 0, delta: 0 } },
    stations: st, deadlines: dl, consults: consults, feed: feed.slice(0, 12) };
}
