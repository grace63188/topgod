/* ── 路由（Code.gs 的 dispatch 呼叫） ─────────────────────── */
function lmsDispatch_(req) {
  var act = String(req.action || '');
  LMS_CACHE = {};   /* 每次請求重新讀取（Apps Script 每次執行本來就是新環境；這裡確保同一環境連續呼叫也不會用到舊資料） */
  /* 不讓 lms 動作使既有快照失效（doPost 會看 SNAP_WRITES；空陣列＝不 bump） */
  try { if (typeof SNAP_WRITES === 'object' && SNAP_WRITES && !SNAP_WRITES[act]) SNAP_WRITES[act] = []; } catch (e) { /* ignore */ }
  var fn = lmsActionOf_(act);
  if (!fn) throw apiError('不支援的動作：' + act + '（資料中心未更新）', 'ERR_BAD_ACTION');
  var me = LMS.PUBLIC.indexOf(act) >= 0 ? null : lmsAuth_(req);
  if (me && lmsPrincipalOnly_(act) && me.role !== 'principal') throw apiError('這個動作只有校長可以執行', 'ERR_FORBIDDEN');
  var spec = typeof LMS_SCHEMA === 'object' && LMS_SCHEMA.ACTIONS[act];
  if (me && spec && spec.cap && !LMS_SCHEMA.can(me.role, spec.cap)) throw apiError('這個動作需要「' + LMS_SCHEMA.PERMS[spec.cap].label + '」權限', 'ERR_FORBIDDEN');
  var out = fn(req, me);
  if (typeof lmsCacheBump_ === 'function' && !/^lms_(ping|bootstrap|deadlines|member|catalog|plan_list|plan_get|plan_prev|queue|schedule|preview|test_preview|log|settings_get|tickets|sync_state|ships|todos|nodes_get|templates_get|dump|bm_key_status|bm_key_get|wb_list|review_list|plan_apply_list|audit|class_map|rules|grants|activity|ranking|card_preview|workbench|quota_status|sync_runs|login_google)$/.test(act)) lmsCacheBump_();   /* 2.2.0：寫入後彙總快取失效 */
  if (me && !/^lms_(ping|bootstrap|deadlines|member|catalog|plan_list|plan_get|plan_prev|queue|schedule|preview|test_preview|log|settings_get|tickets|sync_state|ships|todos|nodes_get|templates_get|dump|bm_key_status|bm_key_get|wb_list|review_list|plan_apply_list|audit|class_map|rules|grants|activity|ranking|card_preview|workbench|quota_status|sync_runs)$/.test(act)) lmsAudit_(me, act, req);
  return out;
}

/* ── 登入（3.5.1：Email 驗證碼 token，兩個後台共用；Code.gs requireToken 驗證、auth.gs 權限名單） ───
 *  token 內含 Email 與身分（REQ_ADMIN_USER）；舊管理密碼 token 效期內仍可用（身分由畫面 viewAs 選擇）；Google ID token 登入已移除
 */
function lmsAuth_(req) {
  if (req.idToken && !req.token) throw apiError('Google 登入已停用，請改用 Email 驗證碼登入', 'ERR_AUTH');
  requireToken(req);
  var u = typeof REQ_ADMIN_USER !== 'undefined' ? REQ_ADMIN_USER : null, asA = req.viewAs === 'assistant';
  if (u) {
    var real = u.role === 'assistant' ? 'assistant' : 'principal';
    var via = typeof lmsViaOf_ === 'function' && typeof REQ_ADMIN_SID !== 'undefined' ? lmsViaOf_(REQ_ADMIN_SID) : '';
    return { email: u.email, name: u.name || (real === 'assistant' ? '助教' : '校長'), role: real === 'principal' && asA ? 'assistant' : real, realRole: real, via: via || 'email' };   /* 校長可切換成助教身分檢視；助教固定；2.0.0 以 Google 登入簽發者 via:'google' */
  }
  return { email: '', name: asA ? '助教' : '校長', role: asA ? 'assistant' : 'principal', realRole: 'principal', via: 'token' };
}
/** 權限名單（auth.gs ADMIN_USERS；未部署 auth.gs 時退回 lms 試算表 users） */

}

/** 讀表；raw=true 只讀 LMS 表本身。否則 members 等表空白時回用既有諮詢教材·專屬訂製資料（lms_reuse.gs，唯讀） */
function lmsRows_(name, raw) {
  if (!raw && LMS_CACHE['t_' + name]) return LMS_CACHE['t_' + name];
  if (raw && LMS_CACHE['r_' + name]) return LMS_CACHE['r_' + name];
  var sh = lmsSheet_(name), n = sh.getLastRow() - 1, out = [];
  if (n > 0) {
    var h = LMS.TABLES[name];
    var vals = sh.getRange(2, 1, n, h.length).getValues();
    if (h[1] === 'json') vals.forEach(function (r) { if (r[0] === '' || r[1] === '') return; try { out.push(JSON.parse(r[1])); } catch (e) { /* ignore */ } });

}

var LMS_PUT_LAST = null;

function lmsPut_(name, rows, opt) {
  opt = opt || {};
  LMS_PUT_LAST = { written: 0, same: 0, added: 0, changedIds: [], addedIds: [], prev: {} };
  if (!rows || !rows.length) return 0;
  var sh = lmsSheet_(name), n = sh.getLastRow() - 1, idx = {}, hashAt = {}, now = nowIso();
  if (n > 0) sh.getRange(2, 1, n, 1).getValues().forEach(function (r, i) { idx[String(r[0])] = i + 2; });
  if (opt.skipSame && n > 0) {
    if (String(sh.getRange(1, 4, 1, 1).getValues()[0][0]) !== 'hash') sh.getRange(1, 4, 1, 1).setValues([['hash']]);
    sh.getRange(2, 4, n, 1).getValues().forEach(function (r, i) { hashAt[i + 2] = String(r[0] || ''); });
  }
  var appends = [], updates = [];
  /* 2.2.0：同一批內重複 id 只留最後一筆（官網 study-plans/overview 會把同一位學姐列兩次 → 同一張課表出現兩次；以前第二筆會走到列號 -1 而整批失敗） */
  var lastOf = {}; rows.forEach(function (o, i) { lastOf[String(o.id)] = i; }); rows = rows.filter(function (o, i) { return lastOf[String(o.id)] === i; });
  rows.forEach(function (o) {
    var json = JSON.stringify(o); if (json.length > 49000) throw apiError('單筆資料過大：' + name + '#' + o.id, 'ERR_VALIDATION');
    var at = idx[String(o.id)], h = opt.skipSame ? lmsRowHash_(json) : '';
    if (opt.skipSame && at > 0 && hashAt[at] === h) { LMS_PUT_LAST.same++; return; }
    var row = opt.skipSame ? [String(o.id), json, now, h] : [String(o.id), json, now];
    if (at) { updates.push([at, row]); LMS_PUT_LAST.changedIds.push(String(o.id)); } else { appends.push(row); idx[String(o.id)] = -1; LMS_PUT_LAST.added++; LMS_PUT_LAST.addedIds.push(String(o.id)); }
  });
  updates.sort(function (a, b) { return a[0] - b[0]; });
  /* 有變的列：先讀舊內容（同步差異用），再寫入 */
  if (opt.wantPrev && updates.length) {
    var readRun = function (r0, k) { sh.getRange(r0, 2, k, 1).getValues().forEach(function (v, q) { try { var ob = JSON.parse(v[0]); LMS_PUT_LAST.prev[String(ob.id)] = ob; } catch (e) { /* ignore */ } }); };
    if (updates.length > 300) readRun(2, n); else { var a = 0; while (a < updates.length) { var c = a; while (c + 1 < updates.length && updates[c + 1][0] === updates[c][0] + 1) c++; readRun(updates[a][0], c - a + 1); a = c + 1; } }
  }
  /* 連續列合併寫入，減少呼叫次數 */
  var i = 0, w = opt.skipSame ? 4 : 3;
  while (i < updates.length) {
    var j = i; while (j + 1 < updates.length && updates[j + 1][0] === updates[j][0] + 1) j++;
    sh.getRange(updates[i][0], 1, j - i + 1, w).setValues(updates.slice(i, j + 1).map(function (u) { return u[1]; }));
    i = j + 1;
  }
  if (appends.length) { if (opt.skipSame && String(sh.getRange(1, 4, 1, 1).getValues()[0][0]) !== 'hash') sh.getRange(1, 4, 1, 1).setValues([['hash']]); sh.getRange(sh.getLastRow() + 1, 1, appends.length, w).setValues(appends); }
  LMS_PUT_LAST.written = updates.length + appends.length;
  if (LMS_PUT_LAST.written) { LMS_CACHE['t_' + name] = null; LMS_CACHE['r_' + name] = null; }
  return opt.skipSame ? LMS_PUT_LAST.written : rows.length;
}
