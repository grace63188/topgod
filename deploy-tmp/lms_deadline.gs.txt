      if (!active(c.mem_id) || !c.deadline_at) return;
      push({ id: 'cwork:' + c.id, kind: 'cwork', l1: 'work', l2: 'consult', l3: '', mem_id: c.mem_id, title: (c.title || '輔導作業'), course: c.counsel_title || '一對一諮詢', at: iso(c.deadline_at),
        doneReason: c.status === 'Y' || c.status === '已通過' ? '輔導作業已審核通過' : c.upload_at ? '輔導作業已上傳' : '', vars: { counsel_id: c.counsel_id, pro_id: c.pro_id }, src: 'member-counsel-works#' + c.id });
    });
    /* 官網進修課表堂次 */
    (db.plans || []).forEach(function (pl) {
      if (!active(pl.mem_id) || (pl.status && pl.status !== 'active')) return;
      (pl.tasks || []).forEach(function (t) {
        if (!t.order_deadline) return;
        var p = P[t.product_id] || { name: t.product_name || ('課程 #' + t.product_id) };
        var bought = owned[pl.mem_id] && owned[pl.mem_id][t.product_id];
        push({ id: 'plan:' + (t.id || pl.id + '-' + t.product_id), kind: 'plan', l1: 'work', l2: 'plan', l3: '', mem_id: pl.mem_id, title: p.name + '｜課表下單截止', course: p.name, at: iso(t.order_deadline),
          doneReason: t.status === 'completed' ? '該堂已下單' : bought && parse(bought) >= parse(pl.created_at || 0) ? '該堂已購買' : '', vars: { pro_id: t.product_id }, src: 'study-plans#' + pl.id });
      });
    });
    /* 級別與點數到期 */
    (db.members || []).forEach(function (m) {
      if ((m.status || 'Y') !== 'Y') return;
      if (m.expired_at && m.level > 1) push({ id: 'level:' + m.id, kind: 'level', l1: 'account', l2: 'level', l3: '', mem_id: m.id, title: LEVELS[m.level] + ' 級別到期', course: LEVELS[m.level], at: iso(m.expired_at), doneReason: '', vars: {}, src: 'members#' + m.id });
      if (m.point_expired_at && (+m.points || 0) > 0) push({ id: 'points:' + m.id, kind: 'points', l1: 'account', l2: 'points', l3: '', mem_id: m.id, title: '點數到期（餘 ' + num(m.points) + ' 點）', course: '點數', at: iso(m.point_expired_at), doneReason: '', vars: {}, src: 'members#' + m.id });
    });
    out.sort(function (a, b) { return parse(a.at) - parse(b.at); });
    return out;
  }
  /** 依時間篩選（今天＝0、3、7、14、30 天內；逾期不列入） */
  function inRange(item, days, now) { var l = daysLeft(item.at, now); return l >= 0 && l <= days; }
  /** 分類樹統計：第一層 → 第二層 → 第三層，每節點 { key, name, count, nearest, items } */
  function tree(items) {
    return TAXONOMY.map(function (t1) {
      var a = items.filter(function (x) { return x.l1 === t1.key; });
      var subs = t1.subs.map(function (t2) {
        var b = a.filter(function (x) { return x.l2 === t2.key; });
        var l3 = [];
        if (t2.series) {
          var by = {};
          b.forEach(function (x) { var k = x.l3 || '未分系列'; (by[k] = by[k] || []).push(x); });
          l3 = Object.keys(by).sort().map(function (k) { return node(k, k, by[k]); });
        }
        var n = node(t2.key, t2.name, b); n.l3 = l3; return n;
      }).filter(function (n) { return n.count > 0 || true; });
      var n1 = node(t1.key, t1.name, a); n1.icon = t1.icon; n1.color = t1.color; n1.def = t1.def; n1.subs = subs; return n1;
    });
    function node(key, name, list) { return { key: key, name: name, count: list.length, nearest: list.length ? list.reduce(function (m, x) { return parse(x.at) < parse(m) ? x.at : m; }, list[0].at) : '', items: list }; }
  }

  /* ── 提醒節點 ─────────────────────────────────────────── */
  /**
   * 個人節點：personal[] = { id, mem_id, scope:{ item?:項目 id, kind?:種類 }, nodes:[{ days, time? }] }
   * 項目層級優先於種類層級。回傳 { list, src } 或 null
   */
  function personalFor(personal, item) {
    var mine = (personal || []).filter(function (p) { return p.mem_id === item.mem_id; });
    var byItem = mine.filter(function (p) { return p.scope && p.scope.item === item.id; })[0];
    if (byItem) return { list: byItem.nodes || [], src: 'item', id: byItem.id };
    var byKind = mine.filter(function (p) { return p.scope && !p.scope.item && p.scope.kind === item.kind; })[0];
    return byKind ? { list: byKind.nodes || [], src: 'kind', id: byKind.id } : null;
  }
  function nodeTime(item, days, time, S) {
    if (item.kind === 'ship') return quietRelease(at(addDays(item.at, -days), time || S.sendTime), S.quiet);   // 寄送：寄出後 N 天
    var dl = parse(item.at), t = at(addDays(item.at, -days), time || S.sendTime);
    if (t > dl) t = Math.max(startOfDay(dl), dl - 2 * HOUR);   // 當天截止時間早於發送時間：提早到截止前 2 小時
    return quietRelease(t, S.quiet);
  }
  /**
   * 單一項目的提醒節點（不含合併與每週上限）：
   *  個人節點先跑；個人節點跑完後回到預設節點，早於最後一個個人節點的預設節點標為「由個人節點取代」；
   *  同一天個人節點與預設節點相撞只留一則；已過時點略過；項目已完成則之後全部取消
   */
  function itemNodes(item, ctx) {
    var S = ctx.S, now = ctx.now, log = ctx.sentByNode || {}, paused = ctx.paused || {};
    var defs = (S.offsets[item.kind] || []).map(function (d) { return { days: d, src: 'default' }; });
    var pers = personalFor(ctx.personal, item);
    var list = [];
    if (pers) pers.list.forEach(function (n) { list.push({ days: +n.days, time: n.time, src: 'personal' }); });
    var lastPersonal = -Infinity;
    list.forEach(function (n) { n.t = nodeTime(item, n.days, n.time, S); if (n.t > lastPersonal) lastPersonal = n.t; });
    var perDay = {};
    list.forEach(function (n) { perDay[dayKey(n.t)] = 1; });
    defs.forEach(function (n) {
      n.t = nodeTime(item, n.days, null, S);
      if (pers && n.t <= lastPersonal) { n.skip = '由個人節點取代'; }
      else if (perDay[dayKey(n.t)]) { n.skip = '與個人節點同一天，只發一則'; }
      list.push(n);
    });
    list.sort(function (a, b) { return a.t - b.t; });
    var dlDay = dayKey(item.at);
    return list.map(function (n) {
      var id = item.id + '|' + dlDay + '|' + (n.src === 'personal' ? 'p' : 'd') + n.days;
      var sent = log[id];
      var o = { id: id, item: item.id, mem_id: item.mem_id, kind: item.kind, days: n.days, src: n.src, t: n.t, at: iso(n.t), status: 'pending', reason: '' };
      if (sent) { o.status = 'sent'; o.sentAt = sent.sentAt; o.clickAt = sent.clickAt || ''; o.card = sent.card; return o; }
      if (n.skip) { o.status = 'skipped'; o.reason = n.skip; return o; }
      if (item.done) { o.status = 'skipped'; o.reason = '已完成：' + item.doneReason; return o; }
      if (paused[item.id] || paused['m:' + item.mem_id]) { o.status = 'paused'; o.reason = '已暫停'; return o; }
      if (n.t < startOfDay(now)) { o.status = 'skipped'; o.reason = '時點已過'; return o; }
      if (item.kind !== 'ship' && n.t > parse(item.at)) { o.status = 'skipped'; o.reason = '時點晚於截止'; return o; }
      if (!autoOn(S, item.kind)) { o.status = 'off'; o.reason = '自動提醒未開啟'; return o; }
      return o;
    });
  }
  /** 優先順序（越小越急）：最後一天 ＞ 接近截止 ＞ 期限尚遠 */
  function urgency(item, now) { var l = daysLeft(item.at, now); return l <= 0 ? 0 : l <= 6 ? 1 : 2; }
  /**
   * plan(items, ctx) → { nodes:{ itemId:[node] }, cards:[{ id, mem_id, day, sendAt, nodes, items, merged, capped }], offCards:[同結構，off:true] }
   * ctx：{ S, now, personal, log:[sent card], paused, horizon(天，預設 45) }
   *  同一學姐同一天的待發節點合併成一張卡；每位學姐每週（週一起算）最多 S.weeklyCap 張（已發送的也計入），超過的依急迫度略過
   *  類別開關關閉的節點（status off）不進 cards，另以 offCards 回傳給提醒排程頁顯示「未開啟，開啟後將會發送」
   */
  function planReminders(items, ctx) {
    var S = ctx.S = settingsOf(ctx.S), now = ctx.now = parse(ctx.now != null ? ctx.now : Date.now());
    var sentByNode = {}, weekSent = {};
    (ctx.log || []).forEach(function (c) {
      (c.nodes || []).forEach(function (nid) { sentByNode[nid] = { sentAt: c.sentAt, clickAt: c.clickAt, card: c.id }; });
      if (c.counted !== false && c.sentAt && !c.manualExempt) { var k = c.mem_id + '@' + weekKey(c.sentAt); weekSent[k] = (weekSent[k] || 0) + 1; }
    });
    ctx.sentByNode = sentByNode;
    var horizon = now + (ctx.horizon || 45) * DAY;
    var byItem = {}, groups = {}, offGroups = {};
    var IT = {}; items.forEach(function (it) { IT[it.id] = it; });
    items.forEach(function (it) {
      var ns = itemNodes(it, ctx); byItem[it.id] = ns;
      ns.forEach(function (n) {
        if ((n.status !== 'pending' && n.status !== 'off') || n.t > horizon) return;
        var g = it.mem_id + '@' + dayKey(n.t), bucket = n.status === 'off' ? offGroups : groups;
        (bucket[g] = bucket[g] || []).push(n);
      });
    });
    function toCard(ns0, off) {
      var ns = ns0.sort(function (a, b) { return parse(IT[a.item].at) - parse(IT[b.item].at); });
      var sendAt = Math.min.apply(null, ns.map(function (n) { return n.t; }));
      var its = ns.map(function (n) { return IT[n.item]; });
      return { id: cardId(ns), mem_id: ns[0].mem_id, day: dayKey(sendAt), sendAt: iso(sendAt), t: sendAt, nodes: ns.map(function (n) { return n.id; }), items: its.map(function (x) { return x.id; }),
        merged: ns.length > 1, prio: Math.min.apply(null, its.map(function (x) { return urgency(x, sendAt); })), kind: its[0].kind, off: !!off };
    }
    var cards = Object.keys(groups).map(function (g) { return toCard(groups[g], false); });
    var offCards = Object.keys(offGroups).map(function (g) { return toCard(offGroups[g], true); }).sort(function (a, b) { return a.t - b.t; });
    /* 每週上限：依週分組，已發送數＋本週待發，超過者依急迫度、時間略過 */
    var byWeek = {};
    cards.forEach(function (c) { var k = c.mem_id + '@' + weekKey(c.t); (byWeek[k] = byWeek[k] || []).push(c); });
    Object.keys(byWeek).forEach(function (k) {
      var room = Math.max(0, S.weeklyCap - (weekSent[k] || 0));
      byWeek[k].sort(function (a, b) { return a.prio - b.prio || a.t - b.t; }).forEach(function (c, i) { if (i >= room) c.capped = true; });
    });
    cards.forEach(function (c) {
      c.nodes.forEach(function (nid) {
        var n = byItem[nid.split('|')[0]].filter(function (x) { return x.id === nid; })[0];
        n.card = c.id; if (c.merged) n.mergedWith = c.items.length - 1;
        if (c.capped) { n.status = 'skipped'; n.reason = '超過每週上限（' + S.weeklyCap + ' 則）'; }
      });
    });
    cards = cards.filter(function (c) { return !c.capped; }).sort(function (a, b) { return a.t - b.t; });
    return { nodes: byItem, cards: cards, offCards: offCards };
  }
  function cardId(nodes) { return 'c' + hash(nodes.map(function (n) { return n.id; }).join(',')).slice(0, 12); }
  /** 發送前重新檢查：用最新資料判斷每個項目是否已完成，回傳 { keep:[id], skip:[{id,reason}] } */
  function recheck(itemIds, freshItems) {
    var F = {}; (freshItems || []).forEach(function (x) { F[x.id] = x; });
    var keep = [], skip = [];
    itemIds.forEach(function (id) {
      var x = F[id];
      if (!x) skip.push({ id: id, reason: '同步資料已無此項目（可能已完成或已移除）' });
      else if (x.done) skip.push({ id: id, reason: '已完成：' + x.doneReason });
      else keep.push(id);
    });
    return { keep: keep, skip: skip };
  }
  /**
   * 手動提醒時間判斷：安靜時段內 → { quiet:true, deferTo }；校長可選強制立即（force）
   */
  function manualTiming(now, S, force) {
    S = settingsOf(S); var t = parse(now);
    if (!inQuiet(t, S.quiet)) return { quiet: false, sendAt: iso(t) };
    var rel = quietRelease(t, S.quiet);
    return force ? { quiet: true, forced: true, sendAt: iso(t), deferTo: iso(rel) } : { quiet: true, forced: false, sendAt: iso(rel), deferTo: iso(rel) };
  }

  }

  function hash(s) {   /* FNV-1a 64（兩段 32 位元）：卡片 id 與預覽簽章用；正式簽章在資料中心以 HMAC 產生 */
    var h1 = 0x811c9dc5, h2 = 0x01000193; s = String(s);
    for (var i = 0; i < s.length; i++) { var c = s.charCodeAt(i); h1 ^= c; h1 = Math.imul(h1, 16777619) >>> 0; h2 ^= c + i; h2 = Math.imul(h2, 2246822519) >>> 0; }
    return ('00000000' + h1.toString(16)).slice(-8) + ('00000000' + h2.toString(16)).slice(-8);
  }

  /**
   * 每堂下單截止（日期＋時間）：
   *  every＝每 N 天、range＝區間平均、after＝接續上一階段（上一階段最後一堂＋gap 天）；
   *  單堂手動 manual：shift＝之後順延（以此為新基準）、solo＝只改這堂
   */
  function computeDates(segments) {
    var prevLast = null;
    return (segments || []).map(function (seg, si) {
      var n = seg.items.length;
      var hm = fmtT(seg.start || '') || '23:59';
      var start = seg.startMode === 'after' && si > 0 && prevLast != null ? at(addDays(prevLast, +seg.gap || 0), hm) : parse(seg.start);
      if (n && seg.items[0].manual) start = parse(seg.items[0].manual);
      var every = Math.max(1, Math.min(60, +seg.every || 3));
      var end = seg.mode === 'range' && seg.end ? parse(seg.end) : null;
      var base = start, baseIdx = 0;
      var items = seg.items.map(function (it, i) {
        var auto = end != null && n > 1 ? at(start + Math.round(i * (startOfDay(end) - startOfDay(start)) / DAY / (n - 1)) * DAY, hm) : base + (i - baseIdx) * every * DAY;
        var d = auto;
        if (it.manual && i > 0) { d = parse(it.manual); if (it.manualMode !== 'solo' && end == null) { base = d; baseIdx = i; } }
        var o = {}; for (var k in it) o[k] = it[k]; o.deadline = iso(d); o.auto = iso(auto); return o;
      });
      if (items.length) prevLast = parse(items[items.length - 1].deadline);
      var s2 = {}; for (var k2 in seg) s2[k2] = seg[k2]; s2.items = items; s2.startAt = iso(start); return s2;
    });
  }

  /** 選課列狀態：上過（已購，看得到不能勾＋原因）、已排在其他階段、級別不足 🔒（可排）、可選 */
  function rowState(p, ctx) {

      return el.ok ? { key: 'ok', label: '可選', disabled: false } : { key: 'warn', label: el.reasons.join('、'), disabled: false, warn: true, reasons: el.reasons };
    }
    if (owned[p.id]) return { key: 'owned', label: '已上過（' + fmtD(owned[p.id]) + ' 購買），不能重複排', disabled: true };
    var segs = ctx.segments || [];
    for (var i = 0; i < segs.length; i++) if (i !== ctx.stageIdx && segs[i].items.some(function (it) { return !it.consult && it.ref === p.id; })) return { key: 'staged', label: '已排在' + STAGE_NAMES[i], disabled: true };

  function consultEligible(c, ctx) {
    var out = { ok: true, reasons: [] }, owned = ctx.owned || {}, prods = ctx.productsById || {}, works = ctx.works || [], now = parse(ctx.now || Date.now());
    if (c.rule === 'after') {
      var mains = Object.keys(owned).filter(function (id) { return prods[id] && prods[id].zone === 'main'; });
      if (!mains.length) { out.ok = false; out.reasons.push('需持有正課'); }
      else {
        var passed = mains.some(function (id) { var ws = works.filter(function (w) { return String(w.pro_id) === String(id) && (w.work_type === 'N' || w.work_type === 'C'); }); return ws.length && ws.every(function (w) { return w.status === 'Y'; }); });
        if (!passed) { out.ok = false; out.reasons.push('正課課後作業與輔導單需審核通過'); }
        else if (!works.some(function (w) { return w.counsel_deadline_at && parse(w.counsel_deadline_at) >= now; })) { out.ok = false; out.reasons.push('課後諮詢下單期限已過'); }
      }
    }
    if (c.rule === 'case' && !Object.keys(owned).some(function (id) { return prods[id] && prods[id].zone === 'case'; })) { out.ok = false; out.reasons.push('需有案例專區購買紀錄'); }
    if (c.rule === 'ai' && !Object.keys(owned).some(function (id) { return prods[id] && prods[id].zone === 'ai'; })) { out.ok = false; out.reasons.push('需先購買 AI 教學包'); }
    return out;
  }
  function priceFor(p, level) { if (!p) return 0; if (p.point_levels && p.point_levels[level] != null) return +p.point_levels[level]; return +p.points || 0; }
  /** 規則提醒（只提醒不阻擋）：點數不足、超過級別到期、超過點數（課費）到期、第一堂早於今天、諮詢資格不符 */
  function checkPlan(member, segments, ctx) {
    ctx = ctx || {};
    var now = parse(ctx.now != null ? ctx.now : Date.now()), items = flatItems(segments), alerts = [], byId = ctx.productsById || {};
    if (!items.length) return [{ level: 'err', key: 'empty', text: '課表還沒有任何一堂' }];
    var total = items.reduce(function (a, it) { return a + (it.consult ? (+it.points || 0) : priceFor(byId[it.ref], member.level)); }, 0);
    if (total > (+member.points || 0)) alerts.push({ level: 'warn', key: 'points', text: '點數不足：合計 ' + num(total) + ' 點，餘額 ' + num(member.points) + ' 點，差 ' + num(total - member.points) + ' 點' });
    var last = items.reduce(function (m, x) { return parse(x.deadline) > parse(m.deadline) ? x : m; }, items[0]);
    var first = items.reduce(function (m, x) { return parse(x.deadline) < parse(m.deadline) ? x : m; }, items[0]);
    if (member.expired_at && parse(last.deadline) > parse(member.expired_at)) alerts.push({ level: 'warn', key: 'lvexp', text: '最後一堂（' + fmtD(last.deadline) + '）超過級別到期 ' + fmtD(member.expired_at) });
    if (member.point_expired_at && parse(last.deadline) > parse(member.point_expired_at)) alerts.push({ level: 'warn', key: 'pexp', text: '最後一堂（' + fmtD(last.deadline) + '）超過課費（點數）到期 ' + fmtD(member.point_expired_at) });
    if (startOfDay(first.deadline) < startOfDay(now)) alerts.push({ level: 'err', key: 'past', text: '第一堂（' + fmtD(first.deadline) + '）早於今天' });
    var lock = items.filter(function (it) { return !it.consult && byId[it.ref] && (member.level || 1) < (byId[it.ref].min_level || ZONE_MIN[byId[it.ref].zone] || 2); });
    if (lock.length) alerts.push({ level: 'warn', key: 'level', text: lock.length + ' 堂目前級別沒有購課資格（仍可排入）' });
    var cs = items.filter(function (it) { return it.consult; });
    var bad = cs.filter(function (it) { return !consultEligible(it.consultType || {}, ctx).ok || (member.level || 1) < 4; });
    if (bad.length) alerts.push({ level: 'warn', key: 'consult', text: bad.length + ' 堂一對一諮詢資格不符：' + bad.map(function (it) { var r = consultEligible(it.consultType || {}, ctx).reasons; if ((member.level || 1) < 4) r = ['級別需至尊以上'].concat(r); return it.name + '（' + r.join('、') + '）'; }).join('；') });
    if (cs.length) alerts.push({ level: 'info', key: 'cnote', text: cs.length + ' 堂一對一諮詢不寫入官網課表，由助教追蹤期限內是否完成預約' });
    return alerts;
  }
  /** 寫回工單內容：POST /api/admin/study-plans、POST /api/admin/study-plans/tasks（只含課程堂） */
  function writebackRequests(member, segments) {
    var items = flatItems(segments).filter(function (i) { return !i.consult; });
    return [{ method: 'POST', path: '/api/admin/study-plans', body: { member_id: member.id } }].concat(items.map(function (it) {

  /** 備案文字：學姐公告版、後台輸入版 */
  function textOutputs(segments, ctx) {
    var items = flatItems(segments), S = settingsOf(ctx && ctx.S), byId = (ctx && ctx.productsById) || {};
    var a = items.map(function (it) {
      return it.consult ? '一對一諮詢｜' + it.name + ' ' + fmtD(it.deadline) + ' 前完成預約\n' + S.links.corder
        : it.name + ' ' + fmtD(it.deadline) + ' ' + fmtT(it.deadline) + ' 前下單\n' + SITE + '/product/show/' + it.ref;
    }).join('\n\n');
    var cs = items.filter(function (it) { return it.consult; });
    var b = items.filter(function (it) { return !it.consult; }).map(function (it) { var p = byId[it.ref] || {}; return (p.zone === 'main' ? '正課區' : (p.series || p.l2 || zoneName(p.zone))) + '｜' + it.name + '｜' + fmtD(it.deadline) + ' ' + fmtT(it.deadline); }).join('\n') +

  /** 上一版未完成的堂：逐堂選擇延續（帶入新課表）或取消 */
  function carryOver(prevItems, choices) {
    var keep = [], cancel = [];

  /** 依「專區 › 第二層」分組；每門課 { id, name, seq, status, date }；status：done 已完成、going 進行中、late 逾期、owned 已購、none 未購 */
  function learningMap(products, ownedRows, works, now) {
    now = parse(now != null ? now : Date.now());
    var owned = {}; (ownedRows || []).forEach(function (r) { if (!owned[r.pro_id] || parse(r.created_at) < parse(owned[r.pro_id].created_at)) owned[r.pro_id] = r; });
    var W = {}; (works || []).forEach(function (w) { if (w.mem_work_id) (W[w.pro_id] = W[w.pro_id] || []).push(w); });
    var zones = {};
    (products || []).forEach(function (p) {
      if (p.status === 'N' && !owned[p.id]) return;
      var z = p.zone || 'other';
      var key = z === 'main' ? p.l2 : (p.series || p.l2);
      zones[z] = zones[z] || {};
      var g = zones[z][key] = zones[z][key] || [];
      var o = owned[p.id], ws = W[p.id] || [];
      var st = 'none', date = '';
      if (o) {
        date = o.created_at;
        var allDone = ws.length && ws.every(function (w) { return w.status === 'Y'; });
        var late = ws.some(function (w) { return !w.upload_at && w.deadline_at && parse(w.deadline_at) < now; });
        var exp = o.expired_at && parse(o.expired_at) < now;
        st = allDone ? 'done' : late ? 'late' : exp && !ws.length ? 'done' : ws.length ? 'going' : 'owned';
      }
      g.push({ id: p.id, name: p.name, seq: p.seq, status: st, date: date, expired_at: o ? o.expired_at : '' });
    });
    return ZONES.filter(function (z) { return zones[z.key]; }).map(function (z) {
      var rows = Object.keys(zones[z.key]).map(function (k) {
        var list = zones[z.key][k].sort(function (a, b) { return (a.seq || 0) - (b.seq || 0) || String(a.name).localeCompare(String(b.name)); });
        return { name: k, courses: list, owned: list.filter(function (x) { return x.status !== 'none'; }).length, total: list.length };
      }).sort(function (a, b) { return b.owned - a.owned || a.name.localeCompare(b.name); });
      return { key: z.key, name: z.name, rows: rows, owned: rows.reduce(function (a, r) { return a + r.owned; }, 0), total: rows.reduce(function (a, r) { return a + r.total; }, 0) };
    });
  }
  /** 作業準時率：準時（截止前上傳）、逾時（截止後上傳）、逾期（過了截止仍未上傳） */
  function homeworkRing(works, now) {
    now = parse(now != null ? now : Date.now());
    var r = { ontime: [], late: [], overdue: [], open: [] };
    (works || []).forEach(function (w) {
      if (!w.mem_work_id || !w.deadline_at) return;
      if (w.upload_at) (parse(w.upload_at) <= parse(w.deadline_at) ? r.ontime : r.late).push(w);
      else (parse(w.deadline_at) < now ? r.overdue : r.open).push(w);
    });
    return r;
  }
  /** 活躍度（0–100）：近 30 天購課、作業上傳、通知點擊加權；沉睡＝60 天沒有任何紀錄 */
  function activity(m, rows, now) {
    now = parse(now != null ? now : Date.now());
    var since = now - 30 * DAY, ev = [];
    (rows.works || []).forEach(function (w) { if (parse(w.created_at) >= since) ev.push({ t: w.created_at, what: '購課', name: w.pro_name || '', w: 12 }); if (w.upload_at && parse(w.upload_at) >= since) ev.push({ t: w.upload_at, what: '上傳作業', name: w.work_title || '', w: 10 }); });
    (rows.clicks || []).forEach(function (c) { if (parse(c.clickAt) >= since) ev.push({ t: c.clickAt, what: '點擊通知', name: c.title || '', w: 4 }); });
    if (m.last_login && parse(m.last_login) >= since) ev.push({ t: m.last_login, what: '登入官網', name: '', w: 8 });
    var score = Math.min(100, ev.reduce(function (a, e) { return a + e.w; }, 0));
    var lastAny = [m.last_login].concat((rows.works || []).map(function (w) { return w.upload_at || w.created_at; })).map(parse).filter(function (x) { return !isNaN(x); });
    var last = lastAny.length ? Math.max.apply(null, lastAny) : NaN;
    var tier = isNaN(last) || now - last > 60 * DAY ? '沉睡' : score >= 60 ? '亮眼' : score >= 25 ? '穩定' : '待喚醒';
    return { score: score, tier: tier, events: ev.sort(function (a, b) { return parse(b.t) - parse(a.t); }) };
  }

  return {
    VERSION: VERSION, DAY: DAY, HOUR: HOUR, LEVELS: LEVELS, ZONES: ZONES, TAXONOMY: TAXONOMY, KINDS: KINDS, AUTO_KINDS: AUTO_KINDS, CONSULT_TYPES: CONSULT_TYPES, WORK_TYPES: WORK_TYPES, DEFAULTS: DEFAULTS, TEMPLATES: TEMPLATES, BTN_LABEL: BTN_LABEL,
    SITE_RE: SITE_RE, CARRIER_RE: CARRIER_RE, STAGE_NAMES: STAGE_NAMES, MAX_STAGES: MAX_STAGES, PICK_TABS: PICK_TABS,
    parse: parse, fmtD: fmtD, fmtDT: fmtDT, fmtT: fmtT, iso: iso, dayKey: dayKey, weekKey: weekKey, startOfDay: startOfDay, at: at, addDays: addDays, daysLeft: daysLeft, inQuiet: inQuiet, quietRelease: quietRelease,
    merge: merge, settingsOf: settingsOf, autoOn: autoOn, fillUrl: fillUrl, urlIssue: urlIssue, hash: hash, num: num, callName: callName, render: render,
    kindIndex: kindIndex, productInfo: productInfo, zoneName: zoneName, openDaysFor: openDaysFor, levelAt: levelAt, watchExpiry: watchExpiry,
    collect: collect, inRange: inRange, tree: tree, personalFor: personalFor, itemNodes: itemNodes, planReminders: planReminders, recheck: recheck, manualTiming: manualTiming, recipientFor: recipientFor,
    stageOf: stageOf, buildCard: buildCard, flexOf: flexOf, trackedUrl: trackedUrl, linkFor: linkFor, carrierUrl: carrierUrl, shipItems: shipItems, sampleItem: sampleItem, dailySummaryText: dailySummaryText,
    newSegment: newSegment, computeDates: computeDates, flatItems: flatItems, rowState: rowState, consultEligible: consultEligible, priceFor: priceFor, checkPlan: checkPlan,
    writebackRequests: writebackRequests, textOutputs: textOutputs, carryOver: carryOver, learningMap: learningMap, homeworkRing: homeworkRing, activity: activity
  };
});

/** lms_queue { days, includeOff? } → 自動提醒（未來 days 天）＋手動排定，含每張圖卡的完整預覽；includeOff 時連「類別未開啟」的卡一起回（off:true，提醒排程頁用） */
function lmsQueue_(req) {
  var days = Math.max(1, Math.min(45, Number(req.days) || 14)), S = lmsSettings_(), now = Date.now();
  var items = lmsItems_(false), plan = lmsPlan_(items), IT = {}; items.forEach(function (x) { IT[x.id] = x; });
  var checks = {}; lmsRows_('checks').forEach(function (c) { checks[c.id] = c; });
  var mk = function (c) {
    var card = lmsCardFor_(c.items.map(function (i) { return IT[i]; }), c.id, S);
    card.sendAt = c.sendAt; card.nodes = c.nodes; card.source = 'auto'; card.check = checks[c.id] || null; card.off = !!c.off;
    card.nodeInfo = c.nodes.map(function (nid) { var n = (plan.nodes[nid.split('|')[0]] || []).filter(function (x) { return x.id === nid; })[0]; return n ? { id: n.id, src: n.src, days: n.days } : { id: nid }; });
    return card;
  };
  var inDays = function (c) { return LMS_CORE.parse(c.sendAt) <= now + days * 86400000; };
  var auto = plan.cards.filter(inDays).map(mk);
  var off = req.includeOff ? (plan.offCards || []).filter(inDays).map(mk) : [];
  var manual = lmsRows_('queue').filter(function (q) { return q.status === 'scheduled'; }).map(function (q) { var c = q.card; c.sendAt = q.sendAt; c.source = 'manual'; c.queueId = q.id; c.check = checks[c.id] || null; return c; });
  var paused = lmsRows_('paused');
  return { cards: manual.concat(auto), offCards: off, paused: paused, settings: { mode: S.mode, previewKey: S.previewKey, weeklyCap: S.weeklyCap, quiet: S.quiet, autoKinds: S.autoKinds, autoSend: S.autoSend !== false } };
}
/** lms_schedule { days } → 未來 N 天每天會自動發出的提醒（含類別未開啟者），給提醒排程頁 */
function lmsSchedule_(req) {
  var days = Math.max(1, Math.min(30, Number(req.days) || 14));
  var q = lmsQueue_({ days: days, includeOff: true });
  var all = q.cards.concat(q.offCards), byDay = {};
  all.forEach(function (c) { var k = LMS_CORE.dayKey(c.sendAt); (byDay[k] = byDay[k] || []).push(c); });
  var out = []; for (var i = 0; i <= days; i++) { var k = LMS_CORE.dayKey(LMS_CORE.addDays(Date.now(), i)); out.push({ day: k, cards: (byDay[k] || []).sort(function (a, b) { return String(a.sendAt).localeCompare(String(b.sendAt)); }) }); }
  return { days: out, settings: q.settings, summary: { dailySummary: lmsSettings_().dailySummary !== false, lastSummary: (lmsGet_('sync_runs', 'daily') || {}).lastSummary || '' } };
}
function lmsSign_(cardId, kind) { return 'lm' + cardId + (kind ? '-' + kind : '') + '.' + hmacHex('lms.' + cardId + '.' + (kind || '')).slice(0, 12); }

/* ── 發送 ─────────────────────────────────────────────── */
/**
 * lms_send { cards:[{ memId, itemIds, title?, body?, cardId? }] | cardIds:[自動佇列的卡 id], force?, defer?, queueIds? }
 *  發送前一律以最新同步資料重新檢查（已完成的項目自動略過並記錄原因）；試營運只發校長預覽收件人
 *  安靜時段：未 force → 排到 09:00（新的一次性觸發器 lmsDeferredRun）；force（校長）→ 立即
 */
function lmsSend_(req, me) {
  var S = lmsSettings_(), now = Date.now();
  var timing = LMS_CORE.manualTiming(now, S, !!req.force);
  if (req.force && me.role !== 'principal') throw apiError('只有校長可以在安靜時段強制發送', 'ERR_FORBIDDEN');
  var specs = lmsSendSpecs_(req);
  if (!specs.length) throw apiError('沒有要發送的圖卡', 'ERR_VALIDATION');
  if (timing.quiet && !timing.forced) {
    var q = specs.map(function (sp) { var card = lmsCardFor_(sp.items, sp.cardId, S, sp.override); return { id: 'q' + LMS_CORE.hash(card.id + now).slice(0, 11), status: 'scheduled', sendAt: timing.sendAt, by: me.email, at: nowIso(), spec: { itemIds: sp.items.map(function (x) { return x.id; }), cardId: sp.cardId, override: sp.override, nodes: sp.nodes }, card: card }; });
    lmsPut_('queue', q);
    lmsDeferTrigger_(timing.sendAt);
    return { deferred: true, sendAt: timing.sendAt, count: q.length };
  }
  return lmsDeliver_(specs, S, me, { manual: !!req.manual });
}
function lmsSendSpecs_(req) {
  var all = lmsItems_(true), F = {}; all.forEach(function (x) { F[x.id] = x; });
  var specs = [];
  (req.cards || []).forEach(function (c) { specs.push({ items: (c.itemIds || []).map(function (i) { return F[i]; }).filter(Boolean), cardId: c.cardId || ('m' + LMS_CORE.hash((c.itemIds || []).join(',') + Utilities.getUuid()).slice(0, 11)), override: { title: c.title, body: c.body, reminder: !!c.reminder }, nodes: [] }); });
  if (req.cardIds && req.cardIds.length) {
    var plan = lmsPlan_(lmsItems_(false)), want = {}; req.cardIds.forEach(function (i) { want[i] = 1; });
    plan.cards.forEach(function (c) { if (want[c.id]) specs.push({ items: c.items.map(function (i) { return F[i]; }).filter(Boolean), cardId: c.id, override: null, nodes: c.nodes }); });
  }
  (req.queueIds || []).forEach(function (qid) { var q = lmsGet_('queue', qid); if (q && q.status === 'scheduled') specs.push({ items: q.spec.itemIds.map(function (i) { return F[i]; }).filter(Boolean), cardId: q.spec.cardId, override: q.spec.override, nodes: q.spec.nodes || [], queueId: q.id }); });
  return specs.filter(function (s) { return s.items.length; });
}
/** 實際發送：重新檢查 → 組卡 → 共用 @52grace 發送函式（額度保護、Retry-Key、逐人結果）→ 紀錄 */
function lmsDeliver_(specs, S, me, opt) {
  opt = opt || {};
  var fresh = lmsItems_(true), results = [], messages = [], meta = [];
  specs.forEach(function (sp) {
    var rc = LMS_CORE.recheck(sp.items.map(function (x) { return x.id; }), fresh);
    if (!rc.keep.length) { results.push({ cardId: sp.cardId, ok: false, skipped: true, reason: rc.skip.map(function (s) { return s.reason; }).join('；') }); lmsSkipLog_(sp, rc.skip, me); return; }
    var items = sp.items.filter(function (x) { return rc.keep.indexOf(x.id) >= 0; });
    var card = lmsCardFor_(items, sp.cardId, S, sp.override);
    if (!card.to.key) { results.push({ cardId: card.id, ok: false, reason: S.mode === 'live' ? '學姐沒有名冊鍵' : '尚未設定校長預覽收件人' }); return; }
    var bad = LMS_CORE.urlIssue(card.button.url, false); if (card.kind !== 'ship' && bad) { results.push({ cardId: card.id, ok: false, reason: '按鈕網址' + bad }); return; }
    messages.push({ key: card.to.key, name: card.to.name, text: card.altText, flex: card.flex, tooLong: false });
    meta.push({ card: card, sp: sp, skipped: rc.skip });
  });
  var r = lmsPushCards_(meta, messages, me, opt);
  return { sent: r.sent, results: results.concat(r.results), quota: r.quota };
}
/** 共用發送：meta[i]＝{ card, sp, skipped }、messages[i]＝{ key, name, text, flex }；額度保護一次檢查整批 */
function lmsPushCards_(meta, messages, me, opt) {
  opt = opt || {};
  var results = [];
  if (!messages.length) return { sent: 0, results: results };
  if (typeof lineRequireConfigured === 'function') lineRequireConfigured();
  var quota = lineQuota(true);
  lineQuotaGuard_(quota, messages.length);   /* 3.3.7 額度保護（不足時整批不發） */
  var at = nowIso(), logs = [], usedNow = 0;
  meta.forEach(function (x, i) {
    /* 逐張呼叫既有 linePushMessages（試營運時收件人都是校長，逐張才能對應每張卡的結果）；Retry-Key 以卡片 id 衍生，同卡重送不重複計費 */
    var out = linePushMessages([messages[i]], function (m) { return 'lms.' + x.card.id + '.' + m.key; }, '');
    usedNow += out.usedNow;
    var fail = out.failed[0];
    var c = x.card;
    results.push({ cardId: c.id, ok: !fail, reason: fail ? fail.error : '', skippedItems: x.skipped });
    if (!fail) logs.push({ id: c.id, mem_id: c.mem_id, memberName: c.memberName, kind: c.kind, title: c.title, body: c.body, rows: c.rows, items: c.items, nodes: x.sp.nodes || [], sentAt: at, by: me ? me.email : '系統（每日觸發器）',
      to: c.to, trial: c.trial, test: !!c.test, url: c.button.url, tracked: c.button.tracked, secondary: c.secondary, link: c.link, hero: c.hero, btnImg: c.button.img, manual: !!opt.manual, counted: !c.trial && !c.test, clickAt: '', clicks: 0, planId: x.sp.planId || '' });
    if (x.sp.queueId) { var q = lmsGet_('queue', x.sp.queueId); if (q) { q.status = fail ? 'failed' : 'sent'; q.result = fail ? fail.error : at; lmsPut_('queue', [q]); } }
  });
  if (typeof quotaAccount === 'function') quotaAccount('lms', usedNow);
  lmsPut_('log', logs);
  try { if (typeof chatLogAppend === 'function') chatLogAppend(logs.map(function (l) { return { key: l.to.key, dir: 'out', kind: 'lms', text: (l.trial ? '【試營運預覽】' : '') + l.title, ok: true }; })); } catch (e) { /* 對話紀錄失敗不影響發送 */ }
  return { sent: logs.length, results: results, quota: typeof lineQuotaAfter_ === 'function' ? lineQuotaAfter_(quota, usedNow) : null };
}

  var log = lmsGet_('log', cardId);

  if (!log.clickAt) log.clickAt = at;
  log.clicks = (log.clicks || 0) + 1;
  var url = log.url, msg = '';
  if (kind === 'c') url = log.secondary && log.secondary.url;
  if (kind === 'r') {
    var sid = (log.items[0] || '').replace(/^ship:/, ''), ship = lmsGet_('ships', sid);
    if (ship && !ship.confirmed_at && !log.trial) { ship.confirmed_at = at; lmsPut_('ships', [ship]); }
    log.receiptAt = log.receiptAt || at;
    url = lmsSettings_().links.home; msg = '已確認收到，謝謝學姐🤍';
  }

  if (!(LMS_CORE.SITE_RE.test(url || '') || (kind === 'c' && LMS_CORE.CARRIER_RE.test(url || '')))) throw apiError('連結已失效', 'ERR_NOT_FOUND');
  return { url: url, msg: msg };
}

/* ── 觸發器（只新增 lms 自己的，不動既有觸發器） ─────────── */
var LMS_TRIGGERS = { daily: 'lmsDeadlineDaily', scan: 'lmsShipScan', deferred: 'lmsDeferredRun' };
function lmsTriggerState_() {
  try { var t = ScriptApp.getProjectTriggers().map(function (x) { return x.getHandlerFunction(); }); return { daily: t.indexOf(LMS_TRIGGERS.daily) >= 0, scan: t.indexOf(LMS_TRIGGERS.scan) >= 0, deferred: t.filter(function (h) { return h === LMS_TRIGGERS.deferred; }).length }; } catch (e) { return null; }
}
/** lms_trigger_install（校長）：每天 12:00 截止通知、每 30 分鐘掃描包裹異常回報 */
function lmsTriggerInstall_() {
  var st = lmsTriggerState_() || {};
  if (!st.daily) ScriptApp.newTrigger(LMS_TRIGGERS.daily).timeBased().everyDays(1).atHour(12).nearMinute(5).inTimezone('Asia/Taipei').create();
  if (!st.scan) ScriptApp.newTrigger(LMS_TRIGGERS.scan).timeBased().everyMinutes(30).create();
  return lmsTriggerState_();
}
function lmsDeferTrigger_(sendAt) {
  var st = lmsTriggerState_() || {}; if (st.deferred) return;
  ScriptApp.newTrigger(LMS_TRIGGERS.deferred).timeBased().at(new Date(LMS_CORE.parse(sendAt) + 60000)).create();
}
/** 安靜時段延後的手動提醒：到 09:00 發送，完成後移除這個一次性觸發器 */
function lmsDeferredRun() {
  LMS_CACHE = {};
  var S = lmsSettings_(), due = lmsRows_('queue').filter(function (q) { return q.status === 'scheduled' && LMS_CORE.parse(q.sendAt) <= Date.now() + 60000; });
  if (due.length) lmsDeliver_(lmsSendSpecs_({ queueIds: due.map(function (q) { return q.id; }) }), S, null, { manual: true });
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === LMS_TRIGGERS.deferred) ScriptApp.deleteTrigger(t); });
}
/**
 * 每日截止通知（每天 12:00）：今天該發的自動提醒，發送前重新檢查；
 *  試營運：最多發 trialAutoMax 張給校長預覽，其餘記為「試營運未發」；正式：發給學姐；autoSend=false 時只排不發
 */
function lmsDeadlineDaily() {
  LMS_CACHE = {};
  var S = lmsSettings_(), now = Date.now(), today = LMS_CORE.dayKey(now);
  var sum = { sent: [], skipped: [], off: {}, note: '' };
  var finish = function (r) { lmsDailySummary_(today, S, sum); return r; };
  if (LMS_CORE.inQuiet(now, S.quiet)) { sum.note = '執行時間在安靜時段內，今天沒有自動發送'; return finish({ skipped: 'quiet' }); }
  var plan = lmsPlan_(lmsItems_(false));
  var isToday = function (c) { return c.day === today && LMS_CORE.parse(c.sendAt) <= now + 30 * 60000; };
  (plan.offCards || []).filter(isToday).forEach(function (c) { sum.off[c.kind] = (sum.off[c.kind] || 0) + 1; });
  var due = plan.cards.filter(isToday);
  var nameOf = function (c) { return (lmsGet_('members', c.mem_id) || {}).name || ''; };
  if (S.autoSend === false) { due.forEach(function (c) { sum.skipped.push({ kind: c.kind, memberName: nameOf(c), reason: '每日自動發送已關閉（只排不發）' }); }); return finish({ due: due.length, sent: 0 }); }
  if (!due.length) return finish({ due: 0, sent: 0 });
  var specs = lmsSendSpecs_({ cardIds: due.map(function (c) { return c.id; }) });
  if (S.mode !== 'live') {
    var max = Math.max(0, Number(S.trialAutoMax) || 0);
    specs.slice(max).forEach(function (sp) { sum.skipped.push({ kind: sp.items[0] ? sp.items[0].kind : '', memberName: nameOf({ mem_id: sp.items[0] ? sp.items[0].mem_id : '' }), reason: '試營運每日預覽上限 ' + max + ' 張' }); });
    specs = specs.slice(0, max);
  }
  var r = lmsDeliver_(specs, S, null, {});
  var byCard = {}; specs.forEach(function (sp) { byCard[sp.cardId] = sp; });
  (r.results || []).forEach(function (x) { var sp = byCard[x.cardId] || { items: [] }, it = sp.items[0] || {}; var row = { kind: it.kind || '', memberName: nameOf({ mem_id: it.mem_id }) }; if (x.ok) sum.sent.push(row); else { row.reason = x.reason || '發送失敗'; sum.skipped.push(row); } });
  return finish(r);
}
/** 每日排程摘要：天使 OA 給校長（設定 dailySummary，預設開）；同一天只發一次 */
function lmsDailySummary_(day, S, sum) {
  try {
    if (S.dailySummary === false) return;
    var st = lmsGet_('sync_runs', 'daily') || { id: 'daily' };
    if (st.lastSummary === day) return;
    var text = LMS_CORE.dailySummaryText(day, S, sum);
    if (typeof adminPush === 'function') adminPush(text, 'lms', 'lms.daily.' + day, '學院提醒摘要');
    st.lastSummary = day; st.lastText = text.slice(0, 1500); st.at = nowIso();
    lmsPut_('sync_runs', [st]);
  } catch (e) { console.error('lms daily summary ' + e.message); }
}
