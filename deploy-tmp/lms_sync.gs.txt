/* ── 路由（Code.gs 的 dispatch 呼叫） ─────────────────────── */
function lmsDispatch_(req) {
  var act = String(req.action || '');
  LMS_CACHE = {};   /* 每次請求重新讀取（Apps Script 每次執行本來就是新環境；這裡確保同一環境連續呼叫也不會用到舊資料） */
  /* 不讓 lms 動作使既有快照失效（doPost 會看 SNAP_WRITES；空陣列＝不 bump） */
  try { if (typeof SNAP_WRITES === 'object' && SNAP_WRITES && !SNAP_WRITES[act]) SNAP_WRITES[act] = []; } catch (e) { /* ignore */ }
  var fn = LMS_ACTIONS[act];
  if (!fn) throw apiError('不支援的動作：' + act, 'ERR_BAD_ACTION');
  var me = LMS.PUBLIC.indexOf(act) >= 0 ? null : lmsAuth_(req);
  if (me && LMS.PRINCIPAL_ONLY.indexOf(act) >= 0 && me.role !== 'principal') throw apiError('這個動作只有校長可以執行', 'ERR_FORBIDDEN');
  var out = fn(req, me);

  return out;
}

/* ── 登入（Google ID token） ───────────────────────────── */
function lmsAuth_(req) {
  /* 沿用既有後台權杖：用諮詢教材·專屬訂製的管理密碼登入（login 動作）取得的 token，同一套 requireToken 驗證 */
  if (!req.idToken && req.token) {
    requireToken(req);
    var asA = req.viewAs === 'assistant';
    return { email: 'admin-token', name: asA ? '助教（管理密碼登入）' : '校長（管理密碼登入）', role: asA ? 'assistant' : 'principal', realRole: 'principal', via: 'token' };
  }
  var tok = String(req.idToken || '');
  if (!tok || tok.length > 4096) throw apiError('請先用 Google 帳號登入', 'ERR_AUTH');
  var c = CacheService.getScriptCache(), ck = 'lms_auth_' + sha256Hex(tok).slice(0, 32), hit = c.get(ck), info = null;
  if (hit) { try { info = JSON.parse(hit); } catch (e) { info = null; } }
  if (!info) {
    var r = UrlFetchApp.fetch('https://oauth2.googleapis.com/tokeninfo?id_token=' + encodeURIComponent(tok), { muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) throw apiError('Google 登入已失效，請重新登入', 'ERR_AUTH');
    var j = JSON.parse(r.getContentText());
    if (j.aud !== LMS.CLIENT_ID || String(j.email_verified) !== 'true' || Number(j.exp) * 1000 < Date.now()) throw apiError('Google 登入驗證失敗', 'ERR_AUTH');
    info = { email: String(j.email).toLowerCase(), name: String(j.name || ''), exp: Number(j.exp) * 1000 };
    c.put(ck, JSON.stringify(info), Math.max(60, Math.min(3000, Math.floor((info.exp - Date.now()) / 1000))));
  }
  if (info.exp < Date.now()) throw apiError('Google 登入已過期，請重新登入', 'ERR_AUTH');
  var role = lmsRoleOf_(info.email);
  if (!role) throw apiError(info.email + ' 沒有後台權限，請校長在設定頁新增', 'ERR_FORBIDDEN');
  var view = req.viewAs === 'assistant' && role === 'principal' ? 'assistant' : role;   /* 校長可切換成助教身分檢視 */
  return { email: info.email, name: info.name, role: view, realRole: role };
}
function lmsRoleOf_(email) {
  var owner = '';
  try { owner = String(Session.getEffectiveUser().getEmail() || '').toLowerCase(); } catch (e) { owner = ''; }
  if (owner && email === owner) return 'principal';
  var u = lmsRows_('users').filter(function (x) { return String(x.email).toLowerCase() === email; })[0];
  return u ? (u.role === 'principal' ? 'principal' : 'assistant') : '';
}
function lmsUsersSave_(users, me) {
  var sh = lmsSheet_('users'), now = nowIso();
  var rows = users.filter(function (u) { return /^[^@\s]+@[^@\s]+$/.test(String(u.email || '')); }).slice(0, 50).map(function (u) { return [String(u.email).toLowerCase(), u.role === 'principal' ? 'principal' : 'assistant', cleanText(u.name).slice(0, 40), now]; });

  LMS_CACHE.t_users = null;
  return { users: lmsRows_('users') };
}

/* ── 試算表儲存（每列＝ id＋JSON） ──────────────────────── */
var LMS_CACHE = {};
function lmsBookId_() {
  var p = PropertiesService.getScriptProperties(), id = p.getProperty(LMS.BOOK_KEY);
  if (id) return id;
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    id = p.getProperty(LMS.BOOK_KEY);
    if (!id) { var ss = SpreadsheetApp.create(LMS.BOOK_NAME); id = ss.getId(); p.setProperty(LMS.BOOK_KEY, id); }
  } finally { lock.releaseLock(); }
  return id;
}
function lmsBook_() { if (!LMS_CACHE.book) LMS_CACHE.book = SpreadsheetApp.openById(lmsBookId_()); return LMS_CACHE.book; }
function lmsSheet_(name) {
  var ss = lmsBook_(), sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    var h = LMS.TABLES[name]; sh.getRange(1, 1, 1, h.length).setValues([h]).setFontWeight('bold'); sh.setFrozenRows(1);
    var first = ss.getSheetByName('工作表1') || ss.getSheetByName('Sheet1'); if (first && ss.getSheets().length > 1) { try { ss.deleteSheet(first); } catch (e) { /* ignore */ } }
  }
  return sh;
}
function lmsRows_(name) {
  if (LMS_CACHE['t_' + name]) return LMS_CACHE['t_' + name];
  var sh = lmsSheet_(name), n = sh.getLastRow() - 1, out = [];
  if (n > 0) {
    var h = LMS.TABLES[name];
    var vals = sh.getRange(2, 1, n, h.length).getValues();
    if (h[1] === 'json') vals.forEach(function (r) { if (r[0] === '' || r[1] === '') return; try { out.push(JSON.parse(r[1])); } catch (e) { /* ignore */ } });
    else vals.forEach(function (r) { if (r[0] === '') return; var o = {}; h.forEach(function (k, i) { o[k] = r[i]; }); out.push(o); });
  }
  LMS_CACHE['t_' + name] = out;
  return out;
}
function lmsGet_(name, id) { var s = String(id); return lmsRows_(name).filter(function (r) { return String(r.id) === s; })[0] || null; }
/** 依 id 新增或覆寫（同一批次一次寫入） */
function lmsPut_(name, rows) {
  if (!rows || !rows.length) return 0;
  var sh = lmsSheet_(name), n = sh.getLastRow() - 1, idx = {}, now = nowIso();
  if (n > 0) sh.getRange(2, 1, n, 1).getValues().forEach(function (r, i) { idx[String(r[0])] = i + 2; });
  var appends = [], updates = [];
  rows.forEach(function (o) {
    var json = JSON.stringify(o); if (json.length > 49000) throw apiError('單筆資料過大：' + name + '#' + o.id, 'ERR_VALIDATION');
    var row = [String(o.id), json, now];
    if (idx[String(o.id)]) updates.push([idx[String(o.id)], row]); else { appends.push(row); idx[String(o.id)] = -1; }
  });
  updates.sort(function (a, b) { return a[0] - b[0]; });
  /* 連續列合併寫入，減少呼叫次數 */
  var i = 0;
  while (i < updates.length) {
    var j = i; while (j + 1 < updates.length && updates[j + 1][0] === updates[j][0] + 1) j++;
    sh.getRange(updates[i][0], 1, j - i + 1, 3).setValues(updates.slice(i, j + 1).map(function (u) { return u[1]; }));
    i = j + 1;
  }
  if (appends.length) sh.getRange(sh.getLastRow() + 1, 1, appends.length, 3).setValues(appends);
  LMS_CACHE['t_' + name] = null;
  return rows.length;
}
function lmsDel_(name, ids) {
  var set = {}; ids.forEach(function (x) { set[String(x)] = 1; });
  var sh = lmsSheet_(name), n = sh.getLastRow() - 1; if (n <= 0) return 0;
  var col = sh.getRange(2, 1, n, 1).getValues(), k = 0;
  for (var r = n - 1; r >= 0; r--) if (set[String(col[r][0])]) { sh.deleteRow(r + 2); k++; }   /* 只刪 lms 自己的列（個人節點、暫停） */
  LMS_CACHE['t_' + name] = null;
  return k;
}

var LMS_SETTING_KEYS = ['mode', 'previewKey', 'previewName', 'quiet', 'sendTime', 'weeklyCap', 'trialAutoMax', 'offsets', 'links', 'carriers', 'templates', 'writeback', 'autoSend', 'autoKinds', 'dailySummary', 'testRecipients'];
function lmsSettingsSave_(patch, me) {
  var sh = lmsSheet_('settings'), n = sh.getLastRow() - 1, idx = {}, now = nowIso();
  if (n > 0) sh.getRange(2, 1, n, 1).getValues().forEach(function (r, i) { idx[r[0]] = i + 2; });
  Object.keys(patch).forEach(function (k) {
    if (LMS_SETTING_KEYS.indexOf(k) < 0) return;
    var v = patch[k];
    if (k === 'links') Object.keys(v || {}).forEach(function (lk) { var issue = LMS_CORE.urlIssue(String(v[lk] || '').replace(/\{\w+\}/g, '1')); if (issue) throw apiError('官網連結範本「' + lk + '」' + issue, 'ERR_VALIDATION'); });
    if (k === 'weeklyCap') v = Math.max(1, Math.min(7, Number(v) || 3));
    if (k === 'previewKey' && v && !/^[0-9a-f]{16}$/.test(String(v))) throw apiError('預覽收件人鍵格式不正確', 'ERR_VALIDATION');
    if (k === 'autoKinds') { var ak = {}; LMS_CORE.AUTO_KINDS.forEach(function (kk) { ak[kk] = !!(v && v[kk] === true); }); v = ak; }
    if (k === 'testRecipients') v = (Array.isArray(v) ? v : []).slice(0, 4).filter(function (r) { return r && /^[0-9a-f]{16}$/.test(String(r.key || '')); }).map(function (r) { return { role: r.role === 'assistant' ? 'assistant' : 'principal', key: String(r.key), name: cleanText(r.name).slice(0, 40) }; });
    if (k === 'dailySummary') v = v !== false;
    var row = [k, JSON.stringify(v), now, me ? me.email : ''];
    if (idx[k]) sh.getRange(idx[k], 1, 1, 4).setValues([row]); else { sh.appendRow(row); idx[k] = sh.getLastRow(); }
  });
  LMS_CACHE.settings = null; LMS_CACHE.t_settings = null;
  return { settings: lmsSettings_() };
}

  return { kind: kind, on: next[kind], autoKinds: out.settings.autoKinds };
}

/**
 * lms_sync { kind, rows, runId, final?, meta? }：kind ∈ members、products、kinds、works、cworks、counsels、site_plans、ticket_results
 *  欄位先過白名單；members 由 uid 算出名冊鍵（與既有系統相同：SHA-256(正規化 uid) 前 16 碼）後丟掉 uid
 */
function lmsSync_(req, me) {
  var kind = String(req.kind || ''), rows = Array.isArray(req.rows) ? req.rows.slice(0, 1500) : [];
  if (!LMS.FIELDS[kind]) throw apiError('不支援的同步種類：' + kind, 'ERR_VALIDATION');
  var keep = LMS.FIELDS[kind];
  var clean = rows.map(function (r) {
    var o = {};
    keep.forEach(function (k) { if (r[k] !== undefined && r[k] !== null && r[k] !== '') o[k] = r[k]; });
    if (kind === 'members') { o.key = r.uid ? lmsRosterKey_(r.uid) : (r.key || ''); o.id = Number(r.id); }
    if (kind === 'works') o.id = String(r.mem_work_id || ('p' + r.mem_pro_id));
    if (kind === 'products' && o.works) o.works = (o.works || []).map(function (w) { return { id: w.id, work_type: w.work_type, work_title: w.work_title, deadline_limit: w.deadline_limit, deadline_days: w.deadline_days, deadline_date: w.deadline_date }; });
    return o;
  }).filter(function (o) { return o.id != null && o.id !== ''; });
  var n = lmsPut_(kind, clean);
  var st = lmsGet_('sync_runs', 'state') || { id: 'state', counts: {} };
  st.counts = st.counts || {}; st.counts[kind] = lmsRows_(kind).length;
  st.lastRun = nowIso(); st.lastBy = me.email;
  if (req.meta && req.meta.bmVersion) st.bmVersion = String(req.meta.bmVersion).slice(0, 12);
  if (kind === 'works' && req.meta) { if (req.meta.newest) st.worksSince = String(req.meta.newest); if (req.meta.cursor != null) st.worksCursor = Number(req.meta.cursor) || 0; }
  lmsPut_('sync_runs', [st]);
  if (req.final) { lmsPut_('sync_runs', [{ id: 'run_' + (req.runId || Utilities.getUuid()), at: nowIso(), by: me.email, kind: kind, rows: n, meta: req.meta || {} }]); lmsAfterSync_(); }
  return { saved: n, total: st.counts[kind] };
}
/** 名冊鍵：與既有系統一致（trim、小寫後 SHA-256 前 16 碼） */
function lmsRosterKey_(uid) {
  /* 與既有名冊同一套：schema().rosterKey 正規化 → rosterHash（SHA-256 前 16 碼） */
  if (typeof schema === 'function' && typeof rosterHash === 'function') { var k = schema().rosterKey(uid); return k ? rosterHash(k) : ''; }
  return sha256Hex(String(uid).replace(/[\u0000-\u001f\u007f​﻿]/g, '').trim().toLowerCase()).slice(0, 16);
}

  try { CacheService.getScriptCache().remove('lms_items_v'); } catch (e) { /* ignore */ }
}
function lmsDb_() {

  LMS_CACHE.db = { members: lmsRows_('members'), products: lmsRows_('products'), kinds: lmsRows_('kinds'), works: lmsRows_('works'), cworks: lmsRows_('cworks'), counsels: lmsRows_('counsels'), plans: lmsRows_('site_plans'), history: [], watch: [] };
  return LMS_CACHE.db;
}
function lmsBootstrap_(me) {
  var S = lmsSettings_(), st = lmsSyncState_();
  var members = lmsRows_('members').filter(function (m) { return (m.status || 'Y') === 'Y'; }).map(function (m) { return { id: m.id, key: m.key, name: m.name, level: m.level, expired_at: m.expired_at, points: m.points, point_expired_at: m.point_expired_at, last_login: m.last_login, bound: lmsBound_(m.key) }; });
  return { me: me, settings: S, sync: st, members: members, version: LMS.VERSION, core: LMS_CORE.VERSION, book: lmsBookId_(), bookUrl: 'https://docs.google.com/spreadsheets/d/' + lmsBookId_() + '/edit', quota: lmsQuotaSafe_(), triggers: lmsTriggerState_() };
}
function lmsBound_(key) {
  if (!key || typeof lineBindingOf !== 'function') return null;
  try { var b = lineBindingOf(key); return b && b.bound ? (b.issue ? 'issue' : 'ok') : 'no'; } catch (e) { return null; }
}
function lmsQuotaSafe_() { try { return typeof lineQuota === 'function' ? lineQuota(false) : null; } catch (e) { return { error: e.message }; } }
function lmsCatalog_() {
  var db = lmsDb_(), kidx = LMS_CORE.kindIndex(db.kinds);
  return { products: db.products.map(function (p) { var i = LMS_CORE.productInfo(p, kidx); var o = {}; for (var k in p) if (k !== 'works') o[k] = p[k]; o.zone = i.zone; o.l2 = i.l2; o.series = i.series; o.seq = i.seq; o.hasWork = !!(p.works && p.works.length); return o; }), kinds: db.kinds, consultTypes: lmsConsultTypes_() };
}
